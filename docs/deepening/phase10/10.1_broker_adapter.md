# Phase 10.1 完成归档 — 券商接口适配层

## 变更摘要
构建统一的券商接口适配层：支持 CTP/xtquant/恒生/中信/国金等主流券商，统一 Order/Trade/Position/Account 接口，提供异步/同步双模式、限流熔断、断线重连、回调机制。

## 核心实现 (`quant/execution/broker_adapter.py`)

### 1. 统一数据模型

```python
# 核心枚举
class BrokerType(Enum):
    CTP = "ctp"
    XTQUANT = "xtquant"
    HONGSU = "hongsu"
    ZHONGXIN = "zhongxin"
    GUOJIN = "guojin"
    SIMULATOR = "simulator"

class OrderSide(Enum): BUY, SELL
class OrderType(Enum): MARKET, LIMIT, FAK, FOK
class OrderStatus(Enum): PENDING_NEW, SUBMITTED, PARTIAL_FILLED, FILLED, ...
class TimeInForce(Enum): DAY, GTC, IOC, FOK

# 核心数据类
@dataclass
class OrderRequest:
    symbol: str
    side: OrderSide
    quantity: float
    order_type: OrderType = OrderType.LIMIT
    price: float = 0.0
    time_in_force: TimeInForce = TimeInForce.DAY
    account_id: str = ""
    strategy_id: str = ""
    client_order_id: str = field(default_factory=lambda: str(uuid.uuid4())[:16])

@dataclass
class OrderResponse:
    order_id: str
    client_order_id: str
    symbol: str = ""
    status: OrderStatus = OrderStatus.PENDING_NEW
    message: str = ""
    timestamp: datetime = field(default_factory=now_utc)

@dataclass
class Trade:
    trade_id: str
    order_id: str
    client_order_id: str
    symbol: str
    side: OrderSide
    price: float
    quantity: float
    timestamp: datetime
    commission: float = 0.0
    tax: float = 0.0

@dataclass
class Position:
    symbol: str
    long_quantity: float = 0.0
    short_quantity: float = 0.0
    long_avg_price: float = 0.0
    short_avg_price: float = 0.0
    market_value: float = 0.0
    unrealized_pnl: float = 0.0
    realized_pnl: float = 0.0

@dataclass
class Account:
    account_id: str
    total_assets: float = 0.0
    available_cash: float = 0.0
    frozen_cash: float = 0.0
    market_value: float = 0.0
    margin_used: float = 0.0
    margin_available: float = 0.0
```

### 2. 适配器基类 (`BrokerAdapterBase`)

```python
class BrokerAdapterBase(ABC):
    """抽象基类 - 定义标准接口."""
    
    def __init__(self, config: BrokerConfig):
        self.config = config
        self._callbacks = {  # 事件回调
            "on_order": [], "on_trade": [], "on_position": [],
            "on_account": [], "on_error": [], "on_connected": [], "on_disconnected": []
        }
        self._rate_limiter = RateLimiter(...)  # 令牌桶限流
        self._reconnect_task = None
        self._heartbeat_task = None
    
    # 核心抽象方法
    @abstractmethod
    async def _connect_impl(self) -> bool: ...
    @abstractmethod
    async def _submit_order_impl(self, request: OrderRequest) -> OrderResponse: ...
    @abstractmethod
    async def _cancel_order_impl(self, order_id: str) -> bool: ...
    @abstractmethod
    async def _query_positions_impl(self) -> List[Position]: ...
    @abstractmethod
    async def _query_account_impl(self) -> Account: ...
    
    # 公共同步接口
    async def connect(self) -> bool: ...
    async def submit_order(self, request: OrderRequest) -> OrderResponse: ...
    async def cancel_order(self, order_id: str) -> bool: ...
    async def query_positions(self) -> List[Position]: ...
    async def query_account(self) -> Account: ...
    
    # 回调机制
    def on_order(self, cb): self._callbacks["on_order"].append(cb)
    def on_trade(self, cb): self._callbacks["on_trade"].append(cb)
    ...
    
    # 自动重连 + 心跳
    async def _reconnect_loop(self): ...
    async def _heartbeat_loop(self): ...
```
