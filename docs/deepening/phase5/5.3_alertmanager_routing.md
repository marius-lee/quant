# quant/monitoring/alertmanager_config.py
class AlertmanagerConfigGenerator:
    """Alertmanager 配置生成器."""

    def generate_config(self) -> str:
        """生成 alertmanager.yml 配置."""
        return yaml.dump({
            "global": {
                "resolve_timeout": "5m",

                "smtp_from": "alerts@quant.com",
                "smtp_auth_username": "alerts@quant.com",
                "smtp_auth_password": "${SMTP_PASSWORD}",

            "templates": ["/etc/alertmanager/templates/*.tmpl"],

                "group_by": ["alertname", "severity", "component"],

### 3. 告警路由配置
```yaml
route:
  group_by: ['alertname', 'severity', 'component']
  group_wait: 30s
  group_interval: 5m
  repeat_interval: 4h
  receiver: 'default'
  routes:
    - match:
        severity: critical
      receiver: critical-alerts
      continue: true
    - match:
        severity: warning
      receiver: warning-alerts
    - match_re:
        component: factor|data|execution
      receiver: team-quant
```

### 3. 告警抑制规则
```yaml
inhibit_rules:
  - source_match:
      severity: critical
    target_match:
      severity: warning
    equal: ['alertname', 'component']
```

### 4. 静默窗口
```python
def create_silence(matcher: dict, duration: str, author: str, comment: str) -> dict:
    """创建静默窗口."""
    return {
        "matchers": [{"name": k, "value": v, "isRegex": False} for k, v in matcher.items()],
        "startsAt": datetime.utcnow().isoformat() + "Z",
        "endsAt": (datetime.utcnow() + parse_duration(duration)).isoformat() + "Z",
        "createdBy": author,
        "comment": comment,
    }
```
