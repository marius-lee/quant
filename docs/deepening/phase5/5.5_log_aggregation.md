### 1. 结构化日志配置
```python
# quant/monitoring/logging_config.py
import structlog
import logging
import sys
from pythonjsonlogger import jsonlogger

def configure_structured_logging():
    """配置结构化日志."""
    
    # 标准库 logging 配置
    logging.basicConfig(
        format="%(message)s",
        stream=sys.stdout,
        level=logging.INFO,
    )
    
    # structlog 配置
    structlog.configure(
        processors=[
            structlog.stdlib.filter_by_level,
            structlog.stdlib.add_logger_name,
            structlog.stdlib.add_log_level,
            structlog.stdlib.PositionalArgumentsFormatter(),
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.StackInfoRenderer(),
            structlog.processors.format_exc_info,
            structlog.processors.UnicodeDecoder(),
            structlog.processors.JSONRenderer(),  # JSON 格式输出
        ],
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        wrapper_class=structlog.stdlib.BoundLogger,
        cache_logger_on_first_use=True,
    )

def get_logger(name: str) -> structlog.BoundLogger:
    """获取结构化日志记录器."""
    return structlog.get_logger(name)
```

### 3. 结构化日志记录器
```python
# quant/monitoring/logging.py
import structlog
from quant.monitoring.logging_config import get_logger

# 使用示例
logger = get_logger("quant.trading")

# 结构化日志记录
logger.info(
    "trade_executed",
    strategy="momentum",
    symbol="000001",
    side="buy",
    volume=1000,
    price=10.50,
    order_id="123456",
)

# 错误日志
logger.error(
    "order_rejected",
    strategy="mean_reversion",
    symbol="600000",
    reason="insufficient_funds",
    order_id="123456",
    exc_info=True,  # 自动包含堆栈跟踪
)

# 关联 Trace ID
logger.info(
    "order_placed",
    trace_id="abc123",
    span_id="def456",
    order_id="789012",
)
```

### 4. Loki 查询示例 (LogQL)
```logql
# 查询某策略的错误日志
{job="quant", strategy="momentum"} |= "error"

# 查询某时间段的错误率
sum(rate({job="quant"} |= "error" [5m])) / sum(rate({job="quant"}[5m]))

# 关联 Trace ID
{job="quant", trace_id="abc123"}

# 统计错误类型分布
topk(10, sum by (reason) (count_over_time({job="quant"} |= "error" [1h])))
```
