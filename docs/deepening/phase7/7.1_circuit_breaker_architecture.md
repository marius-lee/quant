## 核心架构

### 1. 熔断器基类与接口 (`quant/risk/circuit_breaker.py`)

```python
class CircuitState(Enum):
    CLOSED = "closed"       # 正常运行
    HALF_OPEN = "half_open" # 半开：尝试恢复
    OPEN = "open"           # 熔断开启

class CircuitLevel(Enum):
    ACCOUNT = "account"     # 账户级
    STRATEGY = "strategy"   # 策略级
    SYMBOL = "symbol"       # 品种级
    MARKET = "market"       # 全市场级

@dataclass
class CircuitBreakerConfig:
    name: str
    level: str  # "account" | "strategy" | "symbol" | "market"
    # 触发条件
    failure_threshold: int = 5          # 连续失败次数
    error_rate_threshold: float = 0.5   # 错误率阈值
    timeout_threshold: float = 30.0     # 超时阈值(秒)
    # 恢复条件
    recovery_timeout: float = 60.0      # 熔断后等待恢复时间(秒)
    success_threshold: int = 3          # 半开状态下连续成功次数
    # 作用域
    scope: str = "default"              # 作用域标识
    tags: Dict[str, str] = field(default_factory=dict)
```

### 2. 熔断器基类

```python
class CircuitBreakerBase(ABC):
    """熔断器基类"""
    
    def __init__(self, config: dict):
        self._config = config
        self.state = "closed"  # closed | half_open | open
        self._failure_count = 0
        self._success_count = 0
        self._last_failure_time = None
        self._state_change_time = time.time()
        self._lock = threading.RLock()
        self._callbacks = []
    
    @abstractmethod
    def check_trigger(self, metrics: dict) -> bool:
        """检查是否触发熔断"""
        pass
    
    def record_success(self):
        with self._lock:
            self._failure_count = 0
            self._success_count += 1
            if self.state == "half_open":
                if self._success_count >= self._config.get("success_threshold", 3):
                    self._transition_to("closed")
    
    def record_failure(self, error: Exception = None):
        with self._lock:
            self._failure_count += 1
            self._success_count = 0
            self._last_failure_time = time.time()
            if self._check_trigger():
                self._transition_to("open")
    
    def allow_request(self) -> bool:
        """检查是否允许请求通过"""
        with self._lock:
            if self.state == "closed":
                return True
            if self.state == "open":
                if time.time() - self._state_change_time >= self._config.get("recovery_timeout", 60):
                    self._transition_to("half_open")
                    return True
                return False
            # half_open
            return True
```

### 2. 四种具体熔断器实现

| 熔断器 | 级别 | 触发条件 | 熔断动作 |
|--------|------|----------|----------|
| `AccountCircuitBreaker` | 账户级 | 保证金率<130%、单日亏损>5%、保证金使用率>90% | 停止所有开仓，仅允许平仓 |
| `StrategyCircuitBreaker` | 策略级 | 连续亏损≥5次、回撤>15%、胜率<35% | 停止该策略开仓 |
| `SymbolCircuitBreaker` | 品种级 | 日内波动>8%、价差>50bps、成交量>10倍 | 禁止该品种开仓 |
| `MarketCircuitBreaker` | 全市场 | 大盘跌幅>5%、涨跌幅比<0.2、VIX>50 | 全市场停止开仓，仅允许平仓 |

### 2. 熔断器管理器

```python
class CircuitBreakerManager:
    """熔断器管理器：统一管理所有熔断器"""
    
    def register(self, breaker: CircuitBreakerBase):
        with self._lock:
            self._breakers[breaker.config.name] = breaker
    
    def check_all(self) -> Dict[str, CircuitState]:
        """检查所有熔断器状态"""
        return {name: b.state for name, b in self._breakers.items()}
    
    def check_and_trigger(self) -> List[str]:
        """检查并触发熔断，返回触发的熔断器名称"""
        triggered = []
        for name, breaker in self._breakers.items():
            if breaker.state == "closed" and breaker.allow_request() == False:
                triggered = breaker.config.name
                triggered.append(triggered)
        return triggered
```
