### 2. 实时指标推送服务 (`quant/monitoring/realtime_dashboard.py`)

```python
class RealtimeDashboardServer:
    """实时仪表盘 WebSocket 服务器"""
    
    def __init__(self, host: str = "0.0.0.0", port: int = 8080):
        self.host = host
        self.port = port
        self._clients: Set[WebSocketServerProtocol] = set()
        self._risk_engine = get_risk_engine()
        self._circuit_breaker_mgr = get_circuit_breaker_manager()
        self._running = False
        self._server = None

    async def start(self):
        """启动 WebSocket 服务器"""
        self._server = await websockets.serve(
            self._handle_connection,
            self.host,
            self.port,
            ping_interval=20,
            ping_timeout=10
        )
        # 启动定时推送任务
        asyncio.create_task(self._push_loop())
        logger.info(f"Realtime dashboard server started on {self.host}:{self.port}")
    
    async def _push_loop(self):
        """定时推送实时指标"""
        while self._running:
            await asyncio.sleep(1)  # 1秒推送一次
            try:
                metrics = self._collect_metrics()
                message = json.dumps({
                    "type": "metrics_update",
                    "timestamp": datetime.utcnow().isoformat(),
                    "data": metrics
                })
                # 广播给所有连接的客户端
                if self._clients:
                    await asyncio.gather(*[
                        client.send(message) for client in self._clients
                    ], return_exceptions=True)
            except Exception as e:
                logger.error(f"Push loop error: {e}")
    
    async def _handle_connection(self, websocket, path):
        """处理新连接"""
        self._clients.add(websocket)
        try:
            # 发送初始数据
            metrics = self._collect_metrics()
            await websocket.send(json.dumps({
                "type": "initial",
                "timestamp": datetime.utcnow().isoformat(),
                "data": self._collect_metrics()
            }))
            # 保持连接
            async for message in websocket:
                pass  # 心跳保持
        except websockets.exceptions.ConnectionClosed:
            pass
        finally:
            self._clients.discard(websocket)
    
    def _collect_metrics(self) -> dict:
        """收集当前风控指标"""
        risk_engine = get_risk_engine()
        cb_manager = get_circuit_breaker_manager()
        
        risk_metrics = self._risk_engine.get_risk_metrics()
        cb_status = self._circuit_breaker_mgr.check_all()
        
        return {
            "risk_metrics": risk_metrics,
            "circuit_breakers": {k: v.value for k, v in cb_status.items()},
            "timestamp": datetime.utcnow().isoformat()
        }
```

### 3. WebSocket 客户端示例 (前端集成)

```javascript
// frontend/src/hooks/useRiskMetrics.js
import { useEffect, useState } from 'react';

export function useRiskMetrics() {
  const [metrics, setMetrics] = useState(null);
  const [connected, setConnected] = useState(false);
  
  useEffect(() => {
    const ws = new WebSocket(`ws://${window.location.hostname}:8080/ws/risk`);
    
    ws.onopen = () => setConnected(true);
    ws.onclose = () => setConnected(false);
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type === 'metrics_update' || data.type === 'initial') {
        setMetrics(data.data);
      }
    };
    
    return () => ws.close();
  }, []);
  
  return { metrics, connected };
}
```
