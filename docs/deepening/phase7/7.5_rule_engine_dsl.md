```python
# quant/risk/rule_engine.py
"""声明式风控规则引擎 — YAML/DSL 定义规则，热加载、版本控制、A/B 测试、回测验证"""

from __future__ import annotations
import yaml
import re
import threading
import time
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Dict, List, Any, Optional, Callable, Union
from pathlib import Path
from quant.utils.logger import get_logger

logger = get_logger("risk.rule_engine")


class RuleOperator(Enum):
    """规则操作符"""
    EQ = "eq"          # ==
    NE = "ne"          # !=
    GT = "gt"          # >
    GTE = "gte"        # >=
    LT = "lt"          # <
    LTE = "lte"        # <=
    IN = "in"          # in
    NOT_IN = "not_in"  # not in
    CONTAINS = "contains"     # 字符串包含
    REGEX = "regex"          # 正则匹配


class RuleAction(Enum):
    """规则触发动作"""
    ALERT = "alert"           # 告警
    BLOCK = "block"           # 阻断
    REDUCE = "reduce"         # 降仓
    STOP_OPENING = "stop_opening"  # 停止开仓
    FORCE_CLOSE = "force_close"    # 强平
    DEGRADE = "degrade"       # 降级
    CIRCUIT_BREAK = "circuit_break"  # 熔断
    LOG = "log"              # 仅记录日志


@dataclass
class RuleCondition:
    """单个条件"""
    field: str                    # 字段路径，支持点号分隔如 "position.concentration"
    operator: RuleOperator        # 操作符
    value: Any                    # 比较值
    weight: float = 1.0           # 权重（用于评分模式）\n\n@dataclass
class RiskRule:
    \"\"\"风控规则定义\"\"\"
    rule_id: str
    name: str
    description: str = ""
    enabled: bool = True
    version: str = "1.0.0"
    tags: List[str] = field(default_factory=list)
    
    # 条件组合
    conditions: List[RuleCondition] = field(default_factory=list)
    logic: str = "and"  # "and" | "or" | "custom"
    custom_logic: Optional[str] = None  # 自定义逻辑表达式
    
    # 触发动作
    actions: List[Dict[str, Any]] = field(default_factory=list)
    
    # 元数据
    created_at: datetime = field(default_factory=datetime.utcnow)
    updated_at: datetime = field(default_factory=datetime.utcnow)
    created_by: str = ""
    version_history: List[Dict] = field(default_factory=list)
    
    # 运行时状态
    enabled: bool = True
    hit_count: int = 0
    last_triggered: Optional[datetime] = None
    last_evaluation: Optional[datetime] = None
    
    def evaluate(self, context: Dict[str, Any]) -> List[Dict]:
        \"\"\"评估规则，返回触发的动作列表\"\"\"
        results = []
        for condition in self.conditions:
            if not self._evaluate_condition(condition, context):
                if self.logic == "and":
                    return []  # AND 逻辑：任一不满足即不通过
            else:
                # OR 逻辑或自定义逻辑
                pass
        
        # 所有条件满足（AND）或任一满足（OR）
        if self.logic == "and":
            all_met = all(self._evaluate_condition(c, context) for c in self.conditions)
            if not all_met:
                return []
        elif self.logic == "or":
            any_met = any(self._evaluate_condition(c, context) for c in self.conditions)
            if not any_met:
                return []
        
        # 触发动作
        results = []
        for action in self.actions:
            self.hit_count += 1
            self.last_triggered = datetime.utcnow()
            results.append({
                "rule_id": self.rule_id,
                "rule_name": self.name,
                "action": action,
                "triggered_at": datetime.utcnow().isoformat(),
            })
        return results
    
    def _evaluate_condition(self, condition: RuleCondition, context: Dict) -> bool:
        \"\"\"评估单个条件\"\"\"
        # 从 context 中获取字段值（支持点号路径）
        value = self._get_nested_value(context, condition.field)
        if value is None:
            return False
        
        op = condition.operator
        expected = condition.value
        actual = value
        
        if op == RuleOperator.EQ:
            return actual == expected
        elif op == RuleOperator.NE:
            return actual != expected
        elif op == RuleOperator.GT:
            return actual > expected
        elif op == RuleOperator.GTE:
            return actual >= expected
        elif op == RuleOperator.LT:
            return actual < expected
        elif op == RuleOperator.LTE:
            return actual <= expected
        elif op == RuleOperator.IN:
            return actual in expected
        elif op == RuleOperator.NOT_IN:
            return actual not in expected
        elif op == RuleOperator.CONTAINS:
            return expected in str(actual)
        elif op == RuleOperator.REGEX:
            return bool(re.match(expected, str(actual)))
        return False
    
    def _get_nested_value(self, context: Dict, path: str) -> Any:
        \"\"\"获取嵌套字段值，支持点号路径\"\"\"
        keys = path.split(".")
        value = context
        for key in keys:
            if isinstance(value, dict):
                value = value.get(key)
            else:
                return None
            if value is None:
                return None
        return value


class RuleEngine:
    \"\"\"规则引擎：加载、热加载、评估、版本管理\"\"\"
    
    def __init__(self, config_path: str = "config/risk/rules.yaml"):
        self.config_path = Path(config_path)
        self._rules: Dict[str, RiskRule] = {}
        self._lock = threading.RLock()
        self._version = 0
        self._watch_thread: Optional[threading.Thread] = None
        self._running = False
        
        if self.config_path.exists():
            self._load_rules()
            self._start_hot_reload()
    
    def _load_rules(self):
        \"\"\"从 YAML 加载规则\"\"\"
        if not self.config_path.exists():
            logger.warning(f"Rule config not found: {self.config_path}")
            return
        
        with open(self.config_path) as f:
            data = yaml.safe_load(f)
        
        with self._lock:
            self._rules.clear()
            for rule_data in data.get("rules", []):
                rule = self._parse_rule(rule_data)
                self._rules[rule.rule_id] = rule
            self._version += 1
            logger.info(f"Loaded {len(self._rules)} rules, version {self._version}")
    
    def _parse_rule(self, data: dict) -> RiskRule:
        \"\"\"解析 YAML 规则为 RiskRule 对象\"\"\"
        conditions = []
        for cond in data.get("conditions", []):
            conditions.append(RuleCondition(
                field=cond["field"],
                operator=RuleOperator(cond["operator"]),
                value=cond["value"],
                weight=cond.get("weight", 1.0)
            ))
        
        actions = []
        for action in data.get("actions", []):
            if isinstance(action, str):
                actions.append({"type": action, "params": {}})
            else:
                actions.append(action)
        
        return RiskRule(
            rule_id=data["rule_id"],
            name=data["name"],
            description=data.get("description", ""),
            enabled=data.get("enabled", True),
            version=data.get("version", "1.0.0"),
            tags=data.get("tags", []),
            conditions=conditions,
            logic=data.get("logic", "and"),
            custom_logic=data.get("custom_logic"),
            actions=actions,
            created_by=data.get("created_by", "system"),
        )
    
    def _start_hot_reload(self, interval: int = 30):
        \"\"\"启动热加载监控\"\"\"
        self._running = True
        self._watch_thread = threading.Thread(target=self._watch_loop, args=(interval,), daemon=True)
        self._watch_thread.start()
    
    def _watch_loop(self, interval: int):
        last_mtime = self.config_path.stat().st_mtime if self.config_path.exists() else 0
        while self._running:
            try:
                current_mtime = self.config_path.stat().st_mtime
                if current_mtime != last_mtime:
                    logger.info("Rule config changed, reloading...")
                    self._load_rules()
                    last_mtime = current_mtime
            except Exception as e:
                logger.error(f"Hot reload error: {e}")
            time.sleep(interval)
    
    def evaluate(self, context: Dict[str, Any]) -> List[Dict]:
        \"\"\"评估所有规则，返回触发的动作列表\"\"\"
        with self._lock:
            results = []
            for rule in self._rules.values():
                if not rule.enabled:
                    continue
                results.extend(rule.evaluate(context))
            return results
    
    def get_rule(self, rule_id: str) -> Optional[RiskRule]:
        return self._rules.get(rule_id)
    
    def add_rule(self, rule: RiskRule) -> bool:
        with self._lock:
            if rule.rule_id in self._rules:
                return False
            self._rules[rule.rule_id] = rule
            self._version += 1
            return True
    
    def remove_rule(self, rule_id: str) -> bool:
        with self._lock:
            if rule_id in self._rules:
                del self._rules[rule_id]
                self._version += 1
                return True
            return False
    
    def update_rule(self, rule_id: str, updates: dict) -> bool:
        with self._lock:
            rule = self._rules.get(rule_id)
            if not rule:
                return False
            for key, value in updates.items():
                if hasattr(rule, key):
                    setattr(rule, key, value)
            rule.updated_at = datetime.utcnow()
            rule.version = f"{int(float(rule.version)) + 1}.0"
            rule.version_history.append({
                "version": rule.version,
                "updated_at": datetime.utcnow().isoformat(),
                "changes": updates
            })
            self._version += 1
            return True
    
    def get_all_rules(self) -> List[RiskRule]:
        with self._lock:
            return list(self._rules.values())
    
    def get_version(self) -> int:
        return self._version
    
    def shutdown(self):
        self._running = False
        if self._watch_thread:
            self._watch_thread.join(timeout=5)


# 全局实例
_rule_engine = None

def get_rule_engine(config_path: str = "config/risk/rules.yaml") -> RuleEngine:
    global _rule_engine
    if _rule_engine is None:
        _rule_engine = RuleEngine(config_path)
    return _rule_engine
```

## 4. A/B 测试与版本控制

```python
class ABTestManager:
    \"\"\"A/B 测试管理器\"\"\"
    
    def __init__(self, rule_engine: RuleEngine):
        self.engine = rule_engine
        self.experiments: Dict[str, ABExperiment] = {}
    
    def create_experiment(self, name: str, control_rule: str, treatment_rule: str, 
                          traffic_split: float = 0.5) -> str:
        \"\"\"创建 A/B 实验\"\"\"
        exp_id = f"exp_{int(time.time())}"
        self.experiments[exp_id] = ABExperiment(
            id=exp_id,
            name=name,
            control_rule=control_rule,
            treatment_rule=treatment_rule,
            traffic_split=traffic_split,
            created_at=datetime.utcnow(),
            status="running"
        )
        return exp_id
    
    def evaluate(self, context: dict, experiment_id: str) -> List[Dict]:
        \"\"\"在实验中评估\"\"\"
        exp = self.experiments.get(experiment_id)
        if not exp or exp.status != "running":
            return self.engine.evaluate(context)
        
        # 根据流量分配选择规则版本
        import random
        if random.random() < exp.traffic_split:
            rule = self.engine.get_rule(exp.treatment_rule)
        else:
            rule = self.engine.get_rule(exp.control_rule)
        
        if rule:
            return rule.evaluate(context)
        return []
```
