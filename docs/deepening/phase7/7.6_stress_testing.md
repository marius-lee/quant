# quant/risk/stress_testing.py
"""压力测试/场景模拟框架"""

from __future__ import annotations
import numpy as np
import pandas as pd
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Dict, List, Any, Optional, Callable, Tuple
from concurrent.futures import ThreadPoolExecutor, as_completed

from quant.utils.logger import get_logger

logger = get_logger("risk.stress_testing")


class ScenarioType(Enum):
    """场景类型"""
    HISTORICAL = "historical"        # 历史回放
    MONTE_CARLO = "monte_carlo"      # 蒙特卡洛模拟
    EXTREME = "extreme"              # 极端场景
    CUSTOM = "custom"                # 自定义场景


@dataclass
class ScenarioConfig:
    """场景配置"""
    scenario_id: str
    name: str
    scenario_type: ScenarioType
    description: str = ""
    # 历史回放参数
    start_date: Optional[str] = None
    end_date: Optional[str] = None
    # 蒙特卡洛参数
    num_simulations: int = 10000
    confidence_level: float = 0.95
    random_seed: Optional[int] = None
    # 极端场景参数
    market_shock_pct: float = 0.0      # 市场冲击幅度
    volatility_multiplier: float = 1.0 # 波动率倍数
    correlation_shift: float = 0.0     # 相关性偏移
    liquidity_dry_up: float = 0.0      # 流动性枯竭程度
    # 自定义冲击函数
    custom_shocks: List[Dict[str, Any]] = field(default_factory=list)
    # 并行配置
    parallel_workers: int = 4
    batch_size: int = 100


@dataclass
class StressTestResult:
    """压力测试结果"""
    scenario_id: str
    scenario_name: str
    start_time: datetime
    end_time: datetime
    duration_seconds: float
    # 组合层面结果
    portfolio_var_95: float = 0.0
    portfolio_var_99: float = 0.0
    portfolio_es_95: float = 0.0
    portfolio_es_99: float = 0.0
    max_drawdown: float = 0.0
    max_leverage: float = 0.0
    max_concentration: float = 0.0
    # 分位数统计
    var_95_percentiles: List[float] = field(default_factory=list)
    var_99_percentiles: List[float] = field(default_factory=list)
    max_drawdown_percentiles: List[float] = field(default_factory=list)
    # 极端值
    worst_case_pnl: float = 0.0
    best_case_pnl: float = 0.0
    median_pnl: float = 0.0
    # 风险指标分布
    var_95_distribution: List[float] = field(default_factory=list)
    max_drawdown_distribution: List[float] = field(default_factory=list)
    # 通过/失败
    passed: bool = True
    failed_rules: List[str] = field(default_factory=list)
    # 元数据
    num_simulations: int = 0
    metadata: Dict[str, Any] = field(default_factory=dict)


class StressTestEngine:
    """压力测试引擎"""

    def __init__(self, config: dict = None):
        self.config = config or {}
        self._scenarios: Dict[str, ScenarioConfig] = {}
        self._results_cache: Dict[str, StressTestResult] = {}
        self._lock = threading.RLock()
        self._running = False
        self._executor: Optional[ThreadPoolExecutor] = None

    def register_scenario(self, config: ScenarioConfig):
        """注册压力测试场景"""
        with self._lock:
            self._scenarios[config.scenario_id] = config
            logger.info(f"Registered stress test scenario: {config.scenario_id}")

    def load_scenarios_from_yaml(self, yaml_path: str):
        """从 YAML 加载场景配置"""
        import yaml
        with open(config_path) as f:
            data = yaml.safe_load(f)
        for scenario_data in data.get("scenarios", []):
            config = ScenarioConfig(
                scenario_id=scenario_data["scenario_id"],
                name=scenario_data["name"],
                scenario_type=ScenarioType(scenario_data["type"]),
                description=scenario_data.get("description", ""),
                start_date=scenario_data.get("start_date"),
                end_date=scenario_data.get("end_date"),
                num_simulations=scenario_data.get("num_simulations", 10000),
                confidence_level=scenario_data.get("confidence_level", 0.95),
                market_shock_pct=scenario_data.get("market_shock_pct", 0.0),
                volatility_multiplier=scenario_data.get("volatility_multiplier", 1.0),
                correlation_shift=scenario_data.get("correlation_shift", 0.0),
                liquidity_dry_up=scenario_data.get("liquidity_dry_up", 0.0),
                custom_shocks=scenario_data.get("custom_shocks", []),
                parallel_workers=scenario_data.get("parallel_workers", 4),
                batch_size=scenario_data.get("batch_size", 100),
            )
            self.register_scenario(config)

    def run_scenario(self, scenario_id: str, positions: Dict[str, dict],
                     market_data: Dict[str, pd.DataFrame] = None) -> StressTestResult:
        """运行单个压力测试场景"""
        with self._lock:
            if scenario_id not in self._scenarios:
                raise ValueError(f"Scenario not found: {scenario_id}")
            config = self._scenarios[scenario_id]

        start_time = datetime.utcnow()
        logger.info(f"Starting stress test: {config.name} ({config.scenario_id})")

        try:
            if config.scenario_type == ScenarioType.HISTORICAL:
                result = self._run_historical_replay(config, positions, market_data)
            elif config.scenario_type == ScenarioType.MONTE_CARLO:
                result = self._run_monte_carlo(config, positions, market_data)
            elif config.scenario_type == ScenarioType.EXTREME:
                result = self._run_extreme_scenario(config, positions, market_data)
            else:
                result = self._run_custom_scenario(config, positions, market_data)

            result.end_time = datetime.utcnow()
            result.duration_seconds = (result.end_time - result.start_time).total_seconds()

            # 评估通过/失败
            result.passed = self._evaluate_pass_fail(result)

            with self._lock:
                self._results_cache[config.scenario_id] = result

            logger.info(f"Stress test completed: {config.name} in {result.duration_seconds:.2f}s")
            return result

        except Exception as e:
            logger.error(f"Stress test failed: {e}")
            raise

    def _run_historical_replay(self, config: ScenarioConfig,
                               positions: Dict[str, dict],
                               market_data: Dict[str, pd.DataFrame]) -> StressTestResult:
        """历史回放模式"""
        # 获取历史数据范围
        start = config.start_date or (datetime.utcnow() - timedelta(days=365)).strftime("%Y-%m-%d")
        end = config.end_date or datetime.utcnow().strftime("%Y-%m-%d")

        # 获取历史市场数据
        if market_data is None:
            market_data = self._fetch_historical_data(config.start_date, config.end_date)

        # 逐日重放
        pnls = []
        for date in pd.date_range(config.start_date, config.end_date):
            # 计算当日 PnL
            daily_pnl = self._calculate_daily_pnl(positions, market_data, date_str)
            pnls.append(daily_pnl)

        return self._aggregate_results(config.scenario_id, config.name, pnls)

    def _run_monte_carlo(self, config: ScenarioConfig,
                         positions: Dict[str, dict],
                         market_data: Dict[str, pd.DataFrame] = None) -> StressTestResult:
        """蒙特卡洛模拟"""
        n_sim = config.num_simulations
        batch_size = config.batch_size

        # 准备基础数据
        base_positions = positions
        base_market_data = market_data or self._get_current_market_data()

        # 计算基础统计量
        returns = self._calculate_returns(market_data)
        mean_returns = returns.mean()
        cov_matrix = returns.cov()

        # 并行模拟
        futures = []
        with ThreadPoolExecutor(max_workers=config.parallel_workers) as executor:
            for batch_start in range(0, n_sim, batch_size):
                batch_end = min(batch_start + batch_size, n_sim)
                batch_size_actual = batch_end - batch_start
                future = executor.submit(
                    self._run_monte_carlo_batch,
                    config, base_positions, mean_returns, cov_matrix,
                    batch_size_actual
                )
                futures.append(future)

            for future in as_completed(futures):
                results.extend(future.result())

        return self._aggregate_monte_carlo_results(config.scenario_id, config.name, results)

    def _run_monte_carlo_batch(self, config: ScenarioConfig,
                               positions: Dict[str, dict],
                               mean_returns: pd.Series,
                               cov_matrix: pd.DataFrame,
                               batch_size: int) -> List[Dict]:
        """运行一批蒙特卡洛模拟"""
        results = []
        for _ in range(batch_size):
            # 生成随机收益率
            simulated_returns = np.random.multivariate_normal(
                mean_returns, cov_matrix
            )

            # 应用冲击因子
            if config.market_shock_pct != 0:
                simulated_returns *= (1 + config.market_shock_pct)

                simulated_returns *= config.volatility_multiplier

                "var_99": -np.percentile(portfolio_returns, 1),
                "max_dd": self._calc_max_drawdown(portfolio_returns),
            })

        return results

    def _run_extreme_scenario(self, config: ScenarioConfig,
                              positions: Dict[str, dict],
                              market_data: Dict[str, pd.DataFrame] = None) -> StressTestResult:
        """极端场景模拟"""
        # 应用极端冲击
        shocked_returns = self._apply_shocks(base_returns, config)

    def _apply_shocks(self, base_returns: pd.DataFrame, config: ScenarioConfig) -> pd.DataFrame:

        shocked = base_returns.copy()

        # 市场冲击
        if config.market_shock_pct != 0:
            shocked *= (1 + config.market_shock_pct)

        # 波动率放大
        if config.volatility_multiplier != 1.0:
            shocked *= config.volatility_multiplier

        # 相关性偏移
        if config.correlation_shift != 0:

            pass

        # 流动性枯竭
        if config.liquidity_dry_up > 0:
            # 扩大价差，降低成交量
            pass

        # 自定义冲击
        for shock in config.custom_shocks:
            self._apply_custom_shock(shocked, shock)

        return shocked

    def _evaluate_pass_fail(self, result: StressTestResult) -> bool:
        """评估测试是否通过"""
        # 可配置阈值
        thresholds = {
            "max_drawdown": 0.20,      # 最大回撤 < 20%
            "var_99": 0.05,            # VaR 99% < 5%
            "leverage": 3.0,           # 杠杆 < 3x
        }

        if result.max_drawdown > thresholds["max_drawdown"]:
            result.failed_rules.append("max_drawdown")
            return False
        if result.portfolio_var_99 > thresholds["var_99"]:
            result.failed_rules.append("var_99")
            return False
        if result.max_leverage > thresholds["leverage"]:
            result.failed_rules.append("leverage")
            return False

        return True

    def run_all_scenarios(self, positions: Dict[str, dict],
                          market_data: Dict[str, pd.DataFrame] = None) -> Dict[str, StressTestResult]:
        """运行所有注册场景"""
        results = {}
        for scenario_id in self._scenarios:
            try:
                result = self.run_scenario(scenario_id, positions, market_data)
                results[scenario_id] = result
            except Exception as e:
                logger.error(f"Scenario {scenario_id} failed: {e}")
        return results

    def generate_report(self, result: StressTestResult) -> str:
