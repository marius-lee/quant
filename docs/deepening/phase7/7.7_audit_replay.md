class CircuitBreakerAuditEvent:

    def __init__(self, event_id: str, event_type: AuditEventType,

class CircuitBreakerAuditor:
    """熔断器审计器 - 记录所有熔断相关事件"""

    def __init__(self, db_path: str = None):
        self.db_path = db_path or str(Path(_require_cfg("data.cdc.state_dir", "/tmp")) / "circuit_breaker_audit.db")
        self._conn = sqlite3.connect(self.db_path, check_same_thread=False)
        self._conn.row_factory = sqlite3.Row
        self._lock = threading.Lock()
        self._init_db()

    def _init_db(self):
        self._conn.executescript("""
            CREATE TABLE IF NOT EXISTS circuit_breaker_audit (
                event_id INTEGER PRIMARY KEY AUTOINCREMENT,
                event_type TEXT NOT NULL,
                breaker_name TEXT NOT NULL,
                breaker_level TEXT NOT NULL,
                scope TEXT NOT NULL,
                old_state TEXT NOT NULL,
                new_state TEXT NOT NULL,
                trigger_reason TEXT,
                trigger_metrics TEXT,
                operator TEXT NOT NULL,
                operator_id TEXT,
                timestamp REAL NOT NULL,
                duration_seconds REAL,
                metadata TEXT
            );
            CREATE INDEX IF NOT EXISTS idx_audit_breaker_name ON circuit_breaker_audit(breaker_name);
            CREATE INDEX IF NOT EXISTS idx_audit_timestamp ON circuit_breaker_audit(timestamp);
            CREATE INDEX IF NOT EXISTS idx_audit_level ON circuit_breaker_audit(breaker_level);
            CREATE INDEX IF NOT EXISTS idx_audit_type ON circuit_breaker_audit(event_type);
        """)
        self._conn.commit()

    def log_event(self, event: CircuitBreakerAuditEvent):
        """记录审计事件"""
        with self._lock:
            self._conn.execute(
                """INSERT INTO circuit_breaker_audit
                   (event_type, breaker_name, breaker_level, scope, old_state, new_state,
                    trigger_reason, trigger_metrics, operator, operator_id, timestamp, duration_seconds, metadata)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?)""",
                (event.event_type.value, event.breaker_name, event.breaker_level, event.scope,
                 event.old_state, event.new_state, event.trigger_reason,
                 json.dumps(event.trigger_metrics), event.operator, event.operator_id,
                 event.timestamp, event.duration_seconds, json.dumps(event.metadata))
            )
            self._conn.commit()

    def log_circuit_open(self, breaker_name: str, level: str, scope: str,
                         trigger_reason: str, trigger_metrics: dict,
                         operator: str = "system", operator_id: str = ""):
        """记录熔断开启"""
        event = CircuitBreakerAuditEvent(
            event_id=f"{int(time.time()*1000)}_{hash(breaker_name)%10000}",
            event_type=AuditEventType.CIRCUIT_OPEN,
            breaker_name=breaker_name,
            breaker_level=level,
            scope=scope,
            old_state="closed",
            new_state="open",
            trigger_reason=trigger_reason,
            trigger_metrics=trigger_metrics,
            operator=operator,
            operator_id=operator_id,
            timestamp=time.time(),
        )
        self.log_event(event)

    def get_events(self,
                   breaker_name: Optional[str] = None,
                   level: Optional[str] = None,
                   start_time: Optional[float] = None,
                   end_time: Optional[float] = None,
                   event_type: Optional[str] = None,
                   limit: int = 100) -> List[dict]:
        """查询审计事件"""
        with self._lock:
            query = "SELECT * FROM circuit_breaker_audit WHERE 1=1"
            params = []

            if breaker_name:
                query += " AND breaker_name = ?"
                params.append(breaker_name)
            if level:
                query += " AND breaker_level = ?"
                params.append(level)
            if start_time:
                query += " AND timestamp >= ?"
                params.append(start_time)
            if end_time:
                query += " AND timestamp <= ?"
                params.append(end_time)
            if event_type:
                query += " AND event_type = ?"
                params.append(event_type)

            query += " ORDER BY timestamp DESC LIMIT ?"
            params.append(limit)

            cursor = self._conn.execute(query, params)
            return [dict(row) for row in cursor.fetchall()]

class CircuitBreakerPostmortem:
    """熔断复盘报告生成器"""

    def __init__(self, auditor: CircuitBreakerAuditor):
        self.auditor = auditor

    def generate_postmortem(self, breaker_name: str,
                           start_time: float, end_time: float) -> dict:
        """生成单个熔断事件的复盘报告"""
        events = self.auditor.get_events(
            breaker_name=breaker_name,
            start_time=start_time,
            end_time=end_time
        )

        if not events:
            return {"error": "No events found"}

        # 分析
        open_events = [e for e in events if e["new_state"] == "open"]
        close_events = [e for e in events if e["new_state"] == "closed"]

        total_duration = 0
        for oe in open_events:
            matching_close = next((c for c in close_events
                                  if c["timestamp"] > oe["timestamp"]), None)
            if matching_close:
                duration = matching_close["timestamp"] - oe["timestamp"]
                total_duration += duration

        # 根因分析
        root_causes = self._analyze_root_causes(events)

        return {
            "breaker_name": breaker_name,
            "period": {"start": start_time, "end": end_time},
            "summary": {
                "total_events": len(events),
                "open_count": len(open_events),
                "close_count": len(close_events),
                "total_downtime_seconds": total_duration,
                "avg_downtime_seconds": total_duration / len(open_events) if open_events else 0,
            },
            "events": events,
            "root_causes": root_causes,
            "recommendations": self._generate_recommendations(root_causes),
            "generated_at": datetime.utcnow().isoformat(),
        }

    def _analyze_root_causes(self, events: List[dict]) -> List[dict]:
        """根因分析"""
        causes = []
        for e in events:
            if e["event_type"] == "circuit_open":
                cause = {
                    "breaker": e["breaker_name"],
                    "time": e["timestamp"],
                    "reason": e["trigger_reason"],
                    "metrics": e["trigger_metrics"],
                    "severity": "high" if e["breaker_level"] in ["account", "market"] else "medium"
                }
                causes.append(cause)
        return sorted(causes, key=lambda x: x["time"], reverse=True)

    def _generate_recommendations(self, root_causes: List[dict]) -> List[str]:
        """生成改进建议"""
        recs = []
        for cause in root_causes:
            if "margin_ratio" in str(cause["metrics"]):
                recs.append("优化保证金监控阈值，建议提前预警阈值至 150%")
            if "drawdown" in str(cause["metrics"]).lower():
                recs.append("增加最大回撤限制，建议单日回撤限制从 5% 降至 3%")
            if "volatility" in str(cause["metrics"]).lower():
                recs.append("增加波动率监控，建议引入 VIX 指标联动熔断")
            if len(cause["metrics"]) == 0:
                recs.append("完善触发指标记录，确保熔断触发时记录完整上下文指标")
        return list(set(recs))  # 去重
