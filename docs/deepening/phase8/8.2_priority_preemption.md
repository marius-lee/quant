### 1. 增强的优先级继承

```python
class PriorityInheritanceManager:
    """增强的优先级继承 - 支持嵌套锁、优先级天花板."""

    def __init__(self):
        self._task_holders: Dict[str, str] = {}  # resource -> task_id
        self._task_waiters: Dict[str, List[Tuple[str, int]]] = defaultdict(list)  # resource -> [(task_id, priority)]
        self._task_base_priority: Dict[str, int] = {}  # task_id -> base_priority
        self._task_inherited_priority: Dict[str, int] = {}  # task_id -> inherited_priority
        self._lock = threading.RLock()
        self._priority_ceiling: Dict[str, int] = {}  # resource -> ceiling_priority

    def set_priority_ceiling(self, resource: str, ceiling: int):
        """设置优先级天花板 - 预防死锁."""
        self._priority_ceiling[resource] = ceiling

    def acquire(self, task_id: str, resource: str, priority: int) -> bool:
        """获取资源，支持优先级天花板协议."""
        with self._lock:
            # 记录基础优先级
            if task_id not in self._task_base_priority:
                self._task_base_priority[task_id] = priority

            holder = self._task_holders.get(resource)
            if holder is None:
                self._task_holders[resource] = task_id
                self._update_effective_priority(task_id)
                return True

            # 检查优先级天花板
            ceiling = self._priority_ceiling.get(resource, float('inf'))
            if priority > ceiling:
                raise RuntimeError(f"Priority ceiling violation: {priority} > {ceiling}")

            # 记录等待者
            self._task_waiters[resource].append((task_id, priority))
            self._task_waiters[resource].sort(key=lambda x: -x[1])  # 高优先级优先

            # 优先级继承：提升持有者到最高等待者优先级
            max_waiter_priority = self._task_waiters[resource][0][1]
            self._task_inherited_priority[holder] = max(
                self._task_inherited_priority.get(holder, 0),
                max_waiter_priority
            )
            self._update_effective_priority(holder)
            return False

    def release(self, task_id: str, resource: str) -> List[str]:
        """释放资源，恢复优先级."""
        with self._lock:
            if self._task_holders.get(resource) != task_id:
                return []

            waiters = self._task_waiters[resource]
            if waiters:
                next_task, next_priority = waiters.pop(0)
                self._task_holders[resource] = next_task
                self._update_effective_priority(next_task)
                # 恢复原持有者优先级
                self._restore_priority(task_id)
                return [next_task]
            else:
                del self._task_holders[resource]
                self._restore_priority(task_id)
                return []

    def _update_effective_priority(self, task_id: str):
        base = self._task_base_priority.get(task_id, 0)
        inherited = self._task_inherited_priority.get(task_id, 0)
        # 有效优先级 = max(基础, 继承)
        return max(base, inherited)

    def _restore_priority(self, task_id: str):
        """恢复任务优先级：移除该资源带来的继承，重新计算."""
        if task_id in self._task_inherited_priority:
            # 重新计算：检查该任务还持有哪些资源，有哪些等待者
            max_inherited = 0
            for resource, holder in self._task_holders.items():
                if holder == task_id:
                    waiters = self._task_waiters.get(resource, [])
                    if waiters:
                        max_inherited = max(max_inherited, waiters[0][1])
            self._task_inherited_priority[task_id] = max_inherited
```

### 2. 增强的抢占管理

```python
class PreemptionManager:
    """增强抢占 - 支持检查点、优雅降级、抢占策略."""

    class PreemptionPolicy(Enum):
        NONE = "none"                    # 不抢占
        CHECKPOINT = "checkpoint"        # 保存检查点后抢占
        GRACEFUL = "graceful"            # 等待任务主动让出
        FORCE = "force"                  # 强制终止

    def __init__(self):
        self._running_tasks: Dict[str, Task] = {}
        self._task_checkpoints: Dict[str, Any] = {}  # task_id -> checkpoint_data
        self._checkpoint_callback: Optional[Callable[[Task], Any]] = None
        self._restore_callback: Optional[Callable[[Task, Any], bool]] = None
        self._preemption_policy = PreemptionPolicy.CHECKPOINT
        self._preemption_callbacks: List[Callable[[Task, Task], None]] = []
        self._lock = threading.RLock()

    def set_checkpoint_callbacks(
        self,
        save: Callable[[Task], Any],
        restore: Callable[[Task, Any], bool]
    ):
        """设置检查点保存/恢复回调."""
        self._checkpoint_callback = save
        self._restore_callback = restore

    def set_preemption_policy(self, policy: PreemptionPolicy):
        self._preemption_policy = policy

    def on_preemption(self, callback: Callable[[Task, Task], None]):
        """注册抢占回调 (preempted_task, preempting_task)."""
        self._preemption_callbacks.append(callback)

    def try_preempt(
        self,
        high_priority_task: Task,
        tenant_quotas: Dict[str, TenantSchedulingQuota],
        strategy: str = "min_victims"
    ) -> List[Task]:
        """智能抢占策略.

        策略:
        - min_victims: 最少受害者
        - min_priority_loss: 最小优先级损失
        - max_resource_gain: 最大资源释放
        """
        with self._lock:
            if self._preemption_policy == PreemptionPolicy.NONE:
                return []

            needed = high_priority_task.resource_demand
            candidates = self._select_victims(high_priority_task, strategy)

            preempted = []
            for task in candidates:
                if self._preemption_policy == PreemptionPolicy.CHECKPOINT:
                    if self._checkpoint_callback:
                        checkpoint = self._checkpoint_callback(task)
                        self._task_checkpoints[task.task_id] = checkpoint
                    else:
                        continue  # 无检查点回调，跳过
                elif self._preemption_policy == PreemptionPolicy.GRACEFUL:
                    # 发送取消信号，等待任务主动退出
                    task.kwargs['_preempted'] = True
                    continue

                preempted.append(task)
                # 触发回调
                for cb in self._preemption_callbacks:
                    try:
                        cb(task, high_priority_task)
                    except Exception as e:
                        logger.error(f"Preemption callback error: {e}")

                if self._resources_freed(preempted, needed):
                    break

            return preempted

    def _select_victims(self, high_priority_task: Task, strategy: str) -> List[Task]:
        """选择抢占受害者."""
        candidates = [
            t for t in self._running_tasks.values()
            if t.tenant_id != high_priority_task.tenant_id
            and t.effective_priority() < high_priority_task.effective_priority()
        ]

        if strategy == "min_victims":
            # 按释放资源量降序，优先抢占大任务
            candidates.sort(key=lambda t: -sum(t.resource_demand.values()))
        elif strategy == "min_priority_loss":
            # 按优先级升序，优先抢占最低优先级
            candidates.sort(key=lambda t: t.effective_priority())
        elif strategy == "max_resource_gain":
            # 计算每个任务能满足的需求比例
            def gain_score(t):
                score = 0
                for res, demand in high_priority_task.resource_demand.items():
                    if res in t.resource_demand:
                        score += min(t.resource_demand[res], demand)
                return score
            candidates.sort(key=lambda t: -gain_score(t))

        return candidates

    def restore_task(self, task: Task) -> bool:
        """从检查点恢复任务."""
        with self._lock:
            checkpoint = self._task_checkpoints.pop(task.task_id, None)
            if checkpoint and self._restore_callback:
                return self._restore_callback(task, checkpoint)
            return False

    def get_preemption_stats(self) -> Dict:
        with self._lock:
            return {
                "running_tasks": len(self._running_tasks),
                "checkpointed_tasks": len(self._task_checkpoints),
                "policy": self._preemption_policy.value,
            }
```

## 使用示例

```python
# 设置检查点回调
def save_checkpoint(task: Task) -> dict:
    return {
        "progress": task.kwargs.get("progress", 0),
        "state": task.kwargs.get("state", {}),
    }

def restore_checkpoint(task: Task, checkpoint: dict) -> bool:
    task.kwargs["progress"] = checkpoint["progress"]
    task.kwargs["state"] = checkpoint["state"]
    return True

preemption = scheduler._preemption
preemption.set_checkpoint_callbacks(save_checkpoint, restore_checkpoint)
preemption.set_preemption_policy(PreemptionManager.PreemptionPolicy.CHECKPOINT)

# 任务中配合检查点
def my_long_task(progress=0, state=None, _preempted=False, **kwargs):
    for i in range(100):
        if _preempted:  # 被抢占时会设为 True
            # 保存进度
            kwargs["progress"] = i
            kwargs["state"] = {"step": i}
            raise PreemptedException("Task preempted")
        # 正常工作...
        time.sleep(0.1)
```
