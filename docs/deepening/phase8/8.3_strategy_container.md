class ContainerRuntime(Enum):

class ContainerStatus(Enum):

    RUNNING = "running"
    PAUSED = "paused"
    STOPPED = "stopped"
    FAILED = "failed"
    REMOVED = "removed"


@dataclass
class ResourceLimits:
    """容器资源限制."""
    cpu_quota: int = 100000        # CPU quota (microseconds per period)
    cpu_period: int = 100000       # CPU period
    cpu_shares: int = 1024         # CPU shares (相对权重)
    memory_limit: str = "1g"       # 内存限制 (如 "1g", "512m")
    memory_swap: str = "2g"        # Swap 限制
    pids_limit: int = 100          # 进程数限制
    gpu_count: int = 0             # GPU 数量
    gpu_devices: List[str] = field(default_factory=list)  # 指定 GPU 设备
    io_weight: int = 500           # IO 权重 (10-1000)
    network_mode: str = "bridge"   # 网络模式

    def to_docker_kwargs(self) -> Dict:
        return {
            "cpu_quota": self.cpu_quota,
            "cpu_period": self.cpu_period,
            "cpu_shares": self.cpu_shares,
            "mem_limit": self.memory_limit,
            "memswap_limit": self.memory_swap,
            "pids_limit": self.pids_limit,
            "device_requests": [
                docker.types.DeviceRequest(count=self.gpu_count, capabilities=[["gpu"]])
            ] if self.gpu_count > 0 else None,
        }

class DockerRuntime(ContainerRuntimeBase):
    """Docker 运行时实现."""

    def __init__(self, docker_url: str = "unix:///var/run/docker.sock"):
        self._client = docker.DockerClient(base_url=docker_url)
        self._containers: Dict[str, ContainerInstance] = {}
        self._lock = threading.RLock()

    def _build_container_config(self, spec: StrategyContainerSpec) -> Dict:
        """构建 Docker 创建参数."""
        labels = {
            "quant.strategy_id": spec.strategy_id,
            "quant.tenant_id": spec.tenant_id,
            "quant.managed": "true",
        }
        labels.update(spec.labels)

        host_config = self._client.api.create_host_config(
            **spec.resource_limits.to_docker_kwargs(),
            network_mode=spec.resource_limits.network_mode,
            read_only=spec.security_policy.read_only_rootfs,
            cap_drop=spec.security_policy.drop_capabilities,
            cap_add=spec.security_policy.add_capabilities,
            security_opt=[
                f"no-new-privileges:{str(spec.security_policy.no_new_privileges).lower()}"
            ] + ([f"seccomp={spec.security_policy.seccomp_profile}"] if spec.security_policy.seccomp_profile else [])
            + ([f"apparmor={spec.security_policy.apparmor_profile}"] if spec.security_policy.apparmor_profile else []),
        )

        return {
            "image": spec.image,
            "command": spec.command,
            "environment": spec.env,
            "working_dir": spec.working_dir,
            "labels": labels,
            "host_config": host_config,
            "volumes": spec.volumes,
            "healthcheck": spec.health_check,
            "user": spec.security_policy.user,
        }

    def create(self, spec: StrategyContainerSpec) -> ContainerInstance:
        with self._lock:
            config = self._build_container_config(spec)
            container = self._client.api.create_container(**config)
            container_id = container["Id"]

            instance = ContainerInstance(
                container_id=container_id,
                spec=spec,
                status=ContainerStatus.CREATED,
            )
            self._containers[container_id] = instance
            logger.info(f"Created container {container_id[:12]} for strategy {spec.strategy_id}")
            return instance

    def start(self, container_id: str) -> bool:
        with self._lock:
            instance = self._containers.get(container_id)
            if not instance:
                return False
            try:
                self._client.api.start(container_id)
                instance.status = ContainerStatus.RUNNING
                instance.started_at = datetime.utcnow()
                return True
            except Exception as e:
                instance.status = ContainerStatus.FAILED
                instance.error = str(e)
                logger.error(f"Failed to start container {container_id[:12]}: {e}")
                return False

    def stop(self, container_id: str, timeout: int = 30) -> bool:
        with self._lock:
            try:
                self._client.api.stop(container_id, timeout=timeout)
                instance = self._containers.get(container_id)
                if instance:
                    instance.status = ContainerStatus.STOPPED
                    instance.finished_at = datetime.utcnow()
                return True
            except Exception as e:
                logger.error(f"Failed to stop container {container_id[:12]}: {e}")
                return False

    def remove(self, container_id: str, force: bool = False) -> bool:
        with self._lock:
            try:
                self._client.api.remove_container(container_id, force=force)
                self._containers.pop(container_id, None)
                return True
            except Exception as e:
                logger.error(f"Failed to remove container {container_id[:12]}: {e}")
                return False

    def get_status(self, container_id: str) -> ContainerStatus:
        with self._lock:
            instance = self._containers.get(container_id)
            if not instance:
                return ContainerStatus.REMOVED
            try:
                inspect = self._client.api.inspect_container(container_id)
                state = inspect["State"]
                if state["Running"]:
                    return ContainerStatus.RUNNING
                elif state["Status"] == "paused":
                    return ContainerStatus.PAUSED
                elif state["ExitCode"] == 0:
                    return ContainerStatus.STOPPED
                else:
                    return ContainerStatus.FAILED
            except Exception:
                return instance.status

    def get_logs(self, container_id: str, tail: int = 100) -> List[str]:
        try:
            logs = self._client.api.logs(container_id, tail=tail, stdout=True, stderr=True)
            return logs.decode("utf-8", errors="replace").splitlines()
        except Exception as e:
            logger.error(f"Failed to get logs for {container_id[:12]}: {e}")
            return []

    def exec(self, container_id: str, cmd: List[str]) -> Tuple[int, str, str]:
        try:
            exec_id = self._client.api.exec_create(container_id, cmd, stdout=True, stderr=True)
            output = self._client.api.exec_start(exec_id)
            inspect = self._client.api.exec_inspect(exec_id)
            return inspect["ExitCode"], output.decode("utf-8", errors="replace"), ""
        except Exception as e:
            return -1, "", str(e)

    def pull_image(self, image: str, auth: Optional[Dict] = None) -> bool:
        """拉取镜像."""
        try:
            for line in self._client.api.pull(image, stream=True, auth_config=auth):
                pass
            return True
        except Exception as e:
            logger.error(f"Failed to pull image {image}: {e}")
            return False

class StrategyContainerManager:
    """策略容器管理器 - 统一入口."""

    def __init__(
        self,
        runtime: ContainerRuntime = ContainerRuntime.DOCKER,
        docker_url: str = "unix:///var/run/docker.sock",
        default_limits: Optional[ResourceLimits] = None,
        default_network: Optional[NetworkPolicy] = None,
        default_security: Optional[SecurityPolicy] = None,
        image_scanner: Optional[ImageSecurityScanner] = None,
    ):
        self.runtime = runtime
        self._runtime_impl = self._create_runtime(runtime, docker_url)
        self._default_limits = default_limits or ResourceLimits()
        self._default_network = default_network or NetworkPolicy()
        self._default_security = default_security or SecurityPolicy()
        self._image_scanner = image_scanner
        self._instances: Dict[str, ContainerInstance] = {}
        self._lock = threading.RLock()

    def _create_runtime(self, runtime: ContainerRuntime, docker_url: str) -> ContainerRuntimeBase:
        if runtime == ContainerRuntime.DOCKER:
            return DockerRuntime(docker_url)
        elif runtime == ContainerRuntime.KUBERNETES:
            return KubernetesRuntime()
        else:
            raise ValueError(f"Unsupported runtime: {runtime}")

    def deploy_strategy(self, spec: StrategyContainerSpec) -> ContainerInstance:
        """部署策略容器."""
        with self._lock:
            # 1. 安全扫描
            if self._image_scanner:
                scan_result = self._image_scanner.scan(spec.image)
                if scan_result.get("summary", {}).get("critical", 0) > 0:
                    raise RuntimeError(f"Image {spec.image} has critical vulnerabilities")

            # 2. 应用默认配置
            if not spec.resource_limits:
                spec.resource_limits = self._default_limits
            if not spec.network_policy:
                spec.network_policy = self._default_network
            if not spec.security_policy:
                spec.security_policy = self._default_security

            # 3. 创建容器
            instance = self._runtime_impl.create(spec)
            self._instances[instance.container_id] = instance

            # 4. 启动
            if not self._runtime_impl.start(instance.container_id):
                raise RuntimeError(f"Failed to start container for {spec.strategy_id}")

            return instance

    def stop_strategy(self, container_id: str, timeout: int = 30) -> bool:
        with self._lock:
            return self._runtime_impl.stop(container_id, timeout)

    def remove_strategy(self, container_id: str) -> bool:
        with self._lock:
            self._runtime_impl.stop(container_id)
            return self._runtime_impl.remove(container_id)

    def get_strategy_status(self, container_id: str) -> ContainerStatus:
        return self._runtime_impl.get_status(container_id)

    def get_strategy_logs(self, container_id: str, tail: int = 100) -> List[str]:
        return self._runtime_impl.get_logs(container_id, tail)

    def list_strategies(self, tenant_id: Optional[str] = None) -> List[ContainerInstance]:
        with self._lock:
            instances = list(self._instances.values())
            if tenant_id:
                instances = [i for i in instances if i.spec.tenant_id == tenant_id]
            return instances


# 全局实例
_container_manager: Optional[StrategyContainerManager] = None


def get_container_manager() -> StrategyContainerManager:
    global _container_manager
    if _container_manager is None:
        _container_manager = StrategyContainerManager()
    return _container_manager


def init_container_manager(**kwargs) -> StrategyContainerManager:
    global _container_manager
    _container_manager = StrategyContainerManager(**kwargs)
    return _container_manager

# 创建容器管理器
manager = init_container_manager(
    runtime=ContainerRuntime.DOCKER,
    default_limits=ResourceLimits(
        cpu_quota=200000,      # 2 CPU cores
        memory_limit="4g",
        pids_limit=200,
    ),
    default_security=SecurityPolicy(
        read_only_rootfs=True,
        no_new_privileges=True,
        drop_capabilities=["ALL"],

    image_scanner=ImageSecurityScanner("trivy"),

spec = StrategyContainerSpec(
    strategy_id="strategy_momentum_v1",
    tenant_id="tenant_a",
    image="quant/strategy-momentum:v1.2.3",
    command=["python", "run_strategy.py", "--date", "2024-01-15"],
    env={
        "STRATEGY_CONFIG": "/config/momentum.yaml",
        "TENANT_ID": "tenant_a",
    },
    volumes={
        "/host/data": {"bind": "/data", "mode": "ro"},
        "/host/config": {"bind": "/config", "mode": "ro"},
    },
    resource_limits=ResourceLimits(
        cpu_quota=400000,      # 4 CPU
        memory_limit="8g",
        gpu_count=1,
    ),
    health_check={
        "test": ["CMD", "curl", "-f", "http://localhost:8080/health"],
        "interval": 30_000_000_000,  # 30s
        "timeout": 10_000_000_000,   # 10s
        "retries": 3,
    },
)

instance = manager.deploy_strategy(spec)
print(f"Deployed: {instance.container_id[:12]}")

# 监控
while True:
    status = manager.get_strategy_status(instance.container_id)
