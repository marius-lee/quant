from __future__ import annotations
import hashlib
import json
import logging

import subprocess
import sys
import tempfile
import threading
import time
import uuid
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum

from typing import Any, Dict, List, Optional, Set, Tuple

from quant.utils.logger import get_logger
from quant.config.constants import _require_cfg

logger = get_logger("tenant.dependency")


class EnvironmentType(Enum):
    """环境类型."""
    VENV = "venv"           # Python venv (轻量)
    CONDA = "conda"         # Conda (支持非 Python 依赖)
    PIXI = "pixi"           # Pixi (现代、快速、跨平台)
    DOCKER = "docker"       # Docker 镜像 (完全隔离)


class DependencySource(Enum):
    """依赖来源."""
    PYPI = "pypi"
    CONDA_FORGE = "conda-forge"
    GITHUB = "github"
    GITLAB = "gitlab"
    LOCAL = "local"
    PRIVATE = "private"


@dataclass
class Dependency:
    """单个依赖项."""
    name: str
    version: str = ""           # 版本约束 (如 ">=1.0,<2.0", "==1.5.3")
    source: DependencySource = DependencySource.PYPI
    extras: List[str] = field(default_factory=list)  # 额外依赖组
    markers: str = ""           # 环境标记 (如 "python_version>='3.10'")
    optional: bool = False

    def to_pip_spec(self) -> str:
        """转换为 pip 规格."""
        spec = self.name
        if self.extras:
            spec += f"[{','.join(self.extras)}]"
        if self.version:
            spec += self.version
        return spec

    def to_conda_spec(self) -> str:
        """转换为 conda 规格."""
        spec = self.name
        if self.version and not self.version.startswith((">=", "<=", ">", "<", "==", "~=")):
            spec += f"={self.version}"
        elif self.version:
            spec += self.version
        return spec


@dataclass
class DependencyLock:
    """依赖锁文件 - 确保可复现构建."""
    strategy_id: str
    environment_type: EnvironmentType
    python_version: str = "3.11"
    dependencies: List[Dependency] = field(default_factory=list)
    resolved: Dict[str, str] = field(default_factory=dict)  # name -> exact_version
    hashes: Dict[str, List[str]] = field(default_factory=dict)  # name -> [sha256]
    created_at: datetime = field(default_factory=datetime.utcnow)
    content_hash: str = ""

    def compute_hash(self) -> str:
        """计算锁文件内容哈希."""
        data = {
            "deps": sorted([f"{d.name}{d.version}" for d in self.dependencies]),
            "python": self.python_version,
        }
        self.content_hash = hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()[:16]
        return self.content_hash


@dataclass
class BuildCacheEntry:
    """构建缓存条目."""
    lock_hash: str
    environment_path: str
    created_at: datetime
    last_used: datetime
    size_mb: float
    hit_count: int = 0


class DependencyResolver:
    """依赖解析器 - 冲突检测与自动解决."""

    def __init__(self, cache_dir: Optional[Path] = None):
        self.cache_dir = cache_dir or Path(_require_cfg("tenant.dependency.cache_dir", ".cache/deps"))
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, BuildCacheEntry] = {}
        self._lock = threading.RLock()

    def resolve(
        self,
        dependencies: List[Dependency],
        python_version: str = "3.11",
        environment_type: EnvironmentType = EnvironmentType.VENV,
        constraints: Optional[Dict[str, str]] = None,
    ) -> DependencyLock:
        """解析依赖，生成锁文件.

        1. 检查缓存
        2. 运行解析器 (pip-tools/conda-lock/pixi)
        3. 冲突检测
        4. 自动解决策略
        """
        # 生成缓存键
        dep_specs = sorted([d.to_pip_spec() for d in dependencies])
        cache_key = hashlib.sha256(
            f"{environment_type.value}:{python_version}:{':'.join(dep_specs)}".encode()
        ).hexdigest()[:32]

        with self._lock:
            # 检查缓存
            if cache_key in self._cache:
                entry = self._cache[cache_key]
                entry.last_used = datetime.utcnow()
                entry.hit_count += 1
                logger.info(f"Cache hit for dependency resolution: {cache_key[:8]}")
                # 这里应该从缓存恢复 lock 文件
                # 简化：重新解析但跳过网络

        # 实际解析
        lock = self._resolve_with_backend(dependencies, python_version, environment_type, constraints)

        # 存入缓存
        with self._lock:
            self._cache[cache_key] = BuildCacheEntry(
                lock_hash=lock.content_hash,
                environment_path="",  # 环境创建后填入
                created_at=datetime.utcnow(),
                last_used=datetime.utcnow(),
                size_mb=0.0,
            )

        return lock

    def _resolve_with_backend(
        self,
        dependencies: List[Dependency],
        python_version: str,
        env_type: EnvironmentType,
        constraints: Optional[Dict[str, str]],
    ) -> DependencyLock:
        """使用后端解析器."""
        if env_type == EnvironmentType.VENV:
            return self._resolve_pip(dependencies, python_version, constraints)
        elif env_type == EnvironmentType.CONDA:
            return self._resolve_conda(dependencies, python_version, constraints)
        elif env_type == EnvironmentType.PIXI:
            return self._resolve_pixi(dependencies, python_version, constraints)
        else:
            raise ValueError(f"Unsupported environment type: {env_type}")

    def _resolve_pip(
        self,
        dependencies: List[Dependency],
        python_version: str,
        constraints: Optional[Dict[str, str]],
    ) -> DependencyLock:
        """使用 pip-tools (pip-compile) 解析."""
        with tempfile.TemporaryDirectory() as tmpdir:
            tmpdir = Path(tmpdir)
            req_file = tmpdir / "requirements.in"
            lock_file = tmpdir / "requirements.txt"

            # 写入 requirements.in
            with open(req_file, "w") as f:
                for dep in dependencies:
                    f.write(dep.to_pip_spec() + "\n")
                if constraints:
                    for name, ver in constraints.items():
                        f.write(f"{name}{ver}\n")

            # 运行 pip-compile
            cmd = [
                sys.executable, "-m", "piptools", "compile",
                "--generate-hashes",
                "--output-file", str(lock_file),
                str(req_file),
            ]
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
            if result.returncode != 0:
                # 尝试冲突解决
                return self._resolve_conflicts_pip(dependencies, python_version, constraints, result.stderr)

            # 解析输出
            lock = DependencyLock(
                strategy_id="",
                environment_type=EnvironmentType.VENV,
                python_version=python_version,
                dependencies=dependencies,
            )
            with open(lock_file) as f:
                for line in f:
                    line = line.strip()
                    if line and not line.startswith("#") and "==" in line:
                        parts = line.split("==")
                        name = parts[0].strip()
                        version_hash = parts[1].split(" \\")
                        version = version_hash[0].strip()
                        lock.resolved[name] = version
                        if len(version_hash) > 1:
                            # 提取 hash
                            for h in version_hash[1:]:
                                if "sha256:" in h:
                                    lock.hashes.setdefault(name, []).append(h.split("sha256:")[1].strip())

            lock.compute_hash()
            return lock

    def _resolve_conflicts_pip(
        self,
        dependencies: List[Dependency],
        python_version: str,
        constraints: Optional[Dict[str, str]],
        error: str,
    ) -> DependencyLock:
        """冲突自动解决策略.

        策略:
        1. 放宽版本约束 (去掉上界)
        2. 移除可选依赖
        3. 使用 --upgrade-strategy=eager
        4. 分组解析 (核心 vs 非核心)
        """
        logger.warning(f"Dependency conflict detected, attempting resolution: {error[:200]}")

        # 策略 1: 放宽版本约束
        relaxed = []
        for dep in dependencies:
            if dep.version and not dep.optional:
                # 移除上界约束
                if "," in dep.version and any(op in dep.version for op in ["<", "<="]):
                    relaxed_version = dep.version.split(",")[0]  # 保留下界
                    relaxed.append(Dependency(
                        name=dep.name,
                        version=relaxed_version,
                        source=dep.source,
                        extras=dep.extras,
                        markers=dep.markers,
                        optional=dep.optional,
                    ))
                else:
                    relaxed.append(dep)
            else:
                relaxed.append(dep)

        # 重试
        return self._resolve_pip(relaxed, python_version, constraints)

    def _resolve_conda(

        dependencies: List[Dependency],
        python_version: str,
        constraints: Optional[Dict[str, str]],
    ) -> DependencyLock:

class EnvironmentBuilder:
    """环境构建器 - 创建隔离环境."""

    def __init__(self, base_dir: Optional[Path] = None):
        self.base_dir = base_dir or Path(_require_cfg("tenant.dependency.env_dir", ".envs/strategies"))
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self._building: Dict[str, threading.Event] = {}

    def build(
        self,
        strategy_id: str,
        lock: DependencyLock,
        force_rebuild: bool = False,
    ) -> Path:
        """构建环境，返回环境路径."""
        env_path = self.base_dir / f"{strategy_id}_{lock.content_hash[:8]}"

        if env_path.exists() and not force_rebuild:
            logger.info(f"Environment already exists: {env_path}")
            return env_path

        # 防止并发构建同一环境
        with threading.Lock():
            if strategy_id not in self._building:
                self._building[strategy_id] = threading.Event()
            event = self._building[strategy_id]

        if event.is_set():
            # 等待其他线程构建完成
            event.wait(timeout=300)
            if env_path.exists():
                return env_path

        try:
            self._do_build(env_path, lock)
            event.set()
            return env_path
        except Exception as e:
            event.set()
            # 清理失败的构建
            if env_path.exists():
                import shutil
                shutil.rmtree(env_path, ignore_errors=True)
            raise

    def _do_build(self, env_path: Path, lock: DependencyLock):
        """实际构建环境."""
        if lock.environment_type == EnvironmentType.VENV:
            self._build_venv(env_path, lock)
        elif lock.environment_type == EnvironmentType.CONDA:
            self._build_conda(env_path, lock)
        elif lock.environment_type == EnvironmentType.PIXI:
            self._build_pixi(env_path, lock)
        else:
            raise ValueError(f"Unsupported environment type: {lock.environment_type}")

    def _build_venv(self, env_path: Path, lock: DependencyLock):
        """构建 venv 环境."""
        import venv
        logger.info(f"Building venv at {env_path}")

        # 创建虚拟环境
        builder = venv.EnvBuilder(with_pip=True)
        builder.create(env_path)

        # 确定 pip 路径
        if sys.platform == "win32":
            pip_path = env_path / "Scripts" / "pip.exe"
            python_path = env_path / "Scripts" / "python.exe"
        else:
            pip_path = env_path / "bin" / "pip"
            python_path = env_path / "bin" / "python"

        # 升级 pip
        subprocess.run([str(pip_path), "install", "-U", "pip"], check=True, timeout=60)

        # 安装精确版本
        if lock.resolved:
            pkgs = [f"{name}=={version}" for name, version in lock.resolved.items()]
            # 使用 --require-hashes 确保完整性
            cmd = [str(pip_path), "install", "--require-hashes"] + pkgs
            if lock.hashes:
                for name, hashes in lock.hashes.items():
                    for h in hashes:
                        cmd.extend(["--hash", f"sha256:{h}"])
            subprocess.run(cmd, check=True, timeout=300)

        logger.info(f"Venv built successfully: {env_path}")

    def _build_conda(self, env_path: Path, lock: DependencyLock):
        """构建 conda 环境."""
        pass

    def _build_pixi(self, env_path: Path, lock: DependencyLock):
        """构建 pixi 环境."""
        pass


class DependencyManager:
    """策略依赖管理器 - 统一入口."""

    def __init__(
        self,
        cache_dir: Optional[Path] = None,
        env_dir: Optional[Path] = None,
        default_env_type: EnvironmentType = EnvironmentType.VENV,
    ):
        self.resolver = DependencyResolver(cache_dir)
        self.builder = EnvironmentBuilder(env_dir)
        self.default_env_type = default_env_type
        self._environments: Dict[str, Path] = {}  # strategy_id -> env_path
        self._lock = threading.RLock()

    def prepare_environment(
        self,
        strategy_id: str,
        dependencies: List[Dependency],
        python_version: str = "3.11",
        env_type: Optional[EnvironmentType] = None,
        force_rebuild: bool = False,
    ) -> Path:
        """准备策略运行环境."""
        env_type = env_type or self.default_env_type

        # 1. 解析依赖生成锁文件
        lock = self.resolver.resolve(
            dependencies=dependencies,
            python_version=python_version,
            environment_type=env_type,
        )
        lock.strategy_id = strategy_id

        # 2. 构建环境
        env_path = self.builder.build(strategy_id, lock, force_rebuild)

        with self._lock:
            self._environments[strategy_id] = env_path

        return env_path

    def get_environment(self, strategy_id: str) -> Optional[Path]:
        with self._lock:
            return self._environments.get(strategy_id)

    def run_in_environment(self, strategy_id: str, command: List[str], **kwargs) -> subprocess.CompletedProcess:
        """在策略环境中运行命令."""
        env_path = self.get_environment(strategy_id)
        if not env_path:
            raise RuntimeError(f"Environment not prepared for {strategy_id}")

        if sys.platform == "win32":
            python = env_path / "Scripts" / "python.exe"
        else:
            python = env_path / "bin" / "python"

        full_cmd = [str(python)] + command
        return subprocess.run(full_cmd, **kwargs)

    def remove_environment(self, strategy_id: str) -> bool:
        with self._lock:
            env_path = self._environments.pop(strategy_id, None)
            if env_path and env_path.exists():
                import shutil
                shutil.rmtree(env_path, ignore_errors=True)
                return True
        return False

    def list_environments(self) -> Dict[str, Dict]:
        with self._lock:
            result = {}
            for strategy_id, env_path in self._environments.items():
                result[strategy_id] = {
                    "path": str(env_path),
                    "exists": env_path.exists(),
                    "size_mb": self._get_dir_size(env_path) if env_path.exists() else 0,
                }
            return result

    def _get_dir_size(self, path: Path) -> float:
        total = 0
        for f in path.rglob("*"):
            if f.is_file():
                total += f.stat().st_size
        return total / (1024 * 1024)


# 全局实例
_dependency_manager: Optional[DependencyManager] = None


def get_dependency_manager() -> DependencyManager:
    global _dependency_manager
    if _dependency_manager is None:
        _dependency_manager = DependencyManager()
    return _dependency_manager


def init_dependency_manager(**kwargs) -> DependencyManager:
    global _dependency_manager
    _dependency_manager = DependencyManager(**kwargs)
    return _dependency_manager
