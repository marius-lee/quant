class ReloadTrigger(Enum):
    """重载触发方式."""
    SIGNAL = "signal"           # SIGHUP/SIGUSR1/SIGUSR2
    FILE_WATCH = "file_watch"   # 文件监控
    API = "api"                 # API 调用
    SCHEDULE = "schedule"       # 定时检查


class DeploymentStrategy(Enum):
    """部署策略."""
    BLUE_GREEN = "blue_green"       # 蓝绿部署
    ROLLING = "rolling"             # 滚动更新
    CANARY = "canary"               # 灰度发布
    RECREATE = "recreate"           # 重建


class StrategyVersionStatus(Enum):
    """版本状态."""
    PENDING = "pending"         # 待部署
    DEPLOYING = "deploying"     # 部署中
    ACTIVE = "active"           # 运行中
    STANDBY = "standby"         # 待机 (蓝绿)
    ROLLED_BACK = "rolled_back" # 已回滚
    FAILED = "failed"           # 部署失败


@dataclass
class StrategyVersion:
    """策略版本."""
    version_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    strategy_id: str = ""
    code_hash: str = ""          # 代码内容哈希
    config_hash: str = ""        # 配置哈希
    dependencies_hash: str = ""  # 依赖锁哈希
    status: StrategyVersionStatus = StrategyVersionStatus.PENDING
    created_at: datetime = field(default_factory=datetime.utcnow)
    deployed_at: Optional[datetime] = None
    rolled_back_at: Optional[datetime] = None
    metadata: Dict = field(default_factory=dict)


@dataclass
class ReloadConfig:
    """热重载配置."""
    strategy_id: str
    trigger: ReloadTrigger = ReloadTrigger.SIGNAL
    signal: int = signal.SIGHUP  # 默认 SIGHUP
    watch_paths: List[str] = field(default_factory=list)
    debounce_seconds: float = 1.0
    deployment_strategy: DeploymentStrategy = DeploymentStrategy.ROLLING
    canary_percentage: float = 10.0  # 灰度百分比
    health_check: Optional[Callable[[], bool]] = None
    health_check_interval: float = 5.0
    health_check_timeout: float = 30.0
    rollback_on_failure: bool = True
    max_versions: int = 10  # 保留版本数


@dataclass
class ReloadEvent:
    """重载事件."""
    event_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    strategy_id: str = ""
    from_version: Optional[str] = None
    to_version: Optional[str] = None
    trigger: ReloadTrigger = ReloadTrigger.API
    started_at: datetime = field(default_factory=datetime.utcnow)
    completed_at: Optional[datetime] = None
    success: bool = False
    error: Optional[str] = None
    duration_ms: float = 0.0


class ModuleReloader:
    """模块热重载器 - 支持增量重载."""

    def __init__(self):
        self._modules: Dict[str, Any] = {}  # module_name -> module
        self._file_mtimes: Dict[str, float] = {}
        self._lock = threading.RLock()

    def load_module(self, module_name: str, file_path: Path) -> Any:
        """加载模块."""
        with self._lock:
            spec = importlib.util.spec_from_file_location(module_name, file_path)
            if spec is None or spec.loader is None:
                raise ImportError(f"Cannot load module from {file_path}")
            module = importlib.util.module_from_spec(spec)
            sys.modules[module_name] = module
            spec.loader.exec_module(module)
            self._modules[module_name] = module
            self._file_mtimes[module_name] = file_path.stat().st_mtime
            return module

    def reload_module(self, module_name: str, file_path: Path) -> Any:
        """热重载模块."""
        with self._lock:
            old_module = self._modules.get(module_name)
            if old_module is None:
                return self.load_module(module_name, file_path)

            # 检查文件是否变更
            current_mtime = file_path.stat().st_mtime
            if current_mtime == self._file_mtimes.get(module_name, 0):
                return old_module  # 无变化

            # 创建新模块
            spec = importlib.util.spec_from_file_location(module_name, file_path)
            if spec is None or spec.loader is None:
                raise ImportError(f"Cannot reload module from {file_path}")
            new_module = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(new_module)

            # 复制状态 (保留非函数/类属性)
            self._copy_state(old_module, new_module)

            # 替换
            sys.modules[module_name] = new_module
            self._modules[module_name] = new_module
            self._file_mtimes[module_name] = current_mtime

            logger.info(f"Hot reloaded module: {module_name}")
            return new_module

    def _copy_state(self, old: Any, new: Any):
        """复制模块状态."""
        for name in dir(old):
            if name.startswith("_"):
                continue
            obj = getattr(old, name)
            # 只复制非模块、非类、非函数的状态
            if not isinstance(obj, (type, type(lambda: None), type(sys))):
                try:
                    setattr(new, name, obj)
                except Exception:
                    pass


class FileWatcher:
    """文件监控器."""

    def __init__(self, debounce: float = 1.0):
        self.debounce = debounce
        self._watched: Dict[str, Dict] = {}  # path -> {callback, last_event}
        self._running = False
        self._thread: Optional[threading.Thread] = None
        self._lock = threading.RLock()

    def watch(self, path: str, callback: Callable[[str], None], recursive: bool = True):
        with self._lock:
            self._watched[path] = {
                "callback": callback,
                "recursive": recursive,
                "last_event": 0.0,
            }

    def unwatch(self, path: str):
        with self._lock:
            self._watched.pop(path, None)

    def start(self):
        if self._running:
            return
        self._running = True
        self._thread = threading.Thread(target=self._watch_loop, daemon=True, name="file-watcher")
        self._thread.start()

    def stop(self):
        self._running = False
        if self._thread:
            self._thread.join(timeout=5)

    def _watch_loop(self):
        # 简化实现：轮询 mtime
        # 生产环境建议使用 watchdog/inotify
        while self._running:
            for path, info in list(self._watched.items()):
                try:
                    p = Path(path)
                    if p.is_file():
                        mtime = p.stat().st_mtime
                        if mtime > info["last_event"] + self.debounce:
                            info["last_event"] = mtime
                            info["callback"](str(p))
                    elif p.is_dir() and info["recursive"]:
                        for f in p.rglob("*.py"):
                            mtime = f.stat().st_mtime
                            if mtime > info["last_event"] + self.debounce:
                                info["last_event"] = mtime
                                info["callback"](str(f))
                except Exception as e:
                    logger.error(f"File watch error for {path}: {e}")
            time.sleep(0.5)


class StrategyHotReloader:
    """策略热重载器."""

    def __init__(self):
        self._configs: Dict[str, ReloadConfig] = {}
        self._versions: Dict[str, List[StrategyVersion]] = defaultdict(list)
        self._active_version: Dict[str, Optional[str]] = {}  # strategy_id -> version_id
        self._module_reloader = ModuleReloader()
        self._file_watcher = FileWatcher()
        self._event_history: List[ReloadEvent] = []
        self._callbacks: Dict[str, List[Callable]] = defaultdict(list)  # strategy_id -> callbacks
        self._lock = threading.RLock()

        # 信号处理
        self._signal_handlers: Dict[int, Callable] = {}
        self._register_signals()

    def _register_signals(self):
        """注册信号处理器."""
        for sig in (signal.SIGHUP, signal.SIGUSR1, signal.SIGUSR2):
            try:
                self._signal_handlers[sig] = signal.signal(sig, self._signal_handler)
            except (ValueError, OSError):
                pass  # 非主线程或不支持

    def _signal_handler(self, signum, frame):
        """信号处理入口."""
        for strategy_id, config in self._configs.items():
            if config.trigger == ReloadTrigger.SIGNAL and config.signal == signum:
                self.trigger_reload(strategy_id, ReloadTrigger.SIGNAL)

    def register_strategy(self, config: ReloadConfig):
        """注册策略热重载配置."""
        with self._lock:
            self._configs[config.strategy_id] = config
            if config.watch_paths:
                for path in config.watch_paths:
                    self._file_watcher.watch(
                        path,
                        lambda p, sid=config.strategy_id: self._on_file_change(sid, p),
                    )
            if config.trigger == ReloadTrigger.FILE_WATCH:
                self._file_watcher.start()

    def _on_file_change(self, strategy_id: str, file_path: str):
        """文件变更回调."""
        config = self._configs.get(strategy_id)
        if not config:
            return
        # 防抖
        time.sleep(config.debounce_seconds)
        self.trigger_reload(strategy_id, ReloadTrigger.FILE_WATCH)

    def add_version(self, strategy_id: str, version: StrategyVersion) -> str:
        """添加新版本."""
        with self._lock:
            self._versions[strategy_id].append(version)
            # 保留最近 N 个版本
            if len(self._versions[strategy_id]) > self._configs.get(strategy_id, ReloadConfig(strategy_id)).max_versions:
                self._versions[strategy_id].pop(0)
            return version.version_id

    def trigger_reload(
        self,
        strategy_id: str,
        trigger: ReloadTrigger = ReloadTrigger.API,
        target_version: Optional[str] = None,
    ) -> ReloadEvent:
        """触发热重载."""
        config = self._configs.get(strategy_id)
        if not config:
            raise ValueError(f"Strategy {strategy_id} not registered for hot reload")

        event = ReloadEvent(
            strategy_id=strategy_id,
            from_version=self._active_version.get(strategy_id),
            trigger=trigger,
        )

        with self._lock:
            self._event_history.append(event)

        try:
            # 确定目标版本
            if target_version:
                version = next((v for v in self._versions[strategy_id] if v.version_id == target_version), None)
                if not version:
                    raise ValueError(f"Version {target_version} not found")
            else:
                # 使用最新 PENDING 版本
                version = next((v for v in reversed(self._versions[strategy_id]) if v.status == StrategyVersionStatus.PENDING), None)
                if not version:
                    raise ValueError("No pending version to deploy")

            # 执行部署策略
            if config.deployment_strategy == DeploymentStrategy.BLUE_GREEN:
                self._deploy_blue_green(config, version, event)
            elif config.deployment_strategy == DeploymentStrategy.ROLLING:
                self._deploy_rolling(config, version, event)
            elif config.deployment_strategy == DeploymentStrategy.CANARY:
                self._deploy_canary(config, version, event)
            else:
                self._deploy_recreate(config, version, event)

            # 更新激活版本
            version.status = StrategyVersionStatus.ACTIVE
            version.deployed_at = datetime.utcnow()
            self._active_version[strategy_id] = version.version_id

            event.to_version = version.version_id
            event.success = True

        except Exception as e:
            event.success = False
            event.error = str(e)
            logger.error(f"Hot reload failed for {strategy_id}: {e}")

            # 回滚
            if config.rollback_on_failure and event.from_version:
                self.rollback(strategy_id, event.from_version)

        finally:
            event.completed_at = datetime.utcnow()
            event.duration_ms = (event.completed_at - event.started_at).total_seconds() * 1000

        # 触发回调
        for cb in self._callbacks[strategy_id]:
            try:
                cb(event)
            except Exception as e:
                logger.error(f"Reload callback error: {e}")

        return event

    def _deploy_blue_green(self, config: ReloadConfig, version: StrategyVersion, event: ReloadEvent):
        """蓝绿部署：新版本启动并健康检查通过后切换."""
        logger.info(f"Blue-green deploy {config.strategy_id} -> {version.version_id}")
        # 1. 启动新版本 (standby)
        version.status = StrategyVersionStatus.STANDBY
        # 2. 健康检查
        self._wait_healthy(config, version)
        # 3. 切换流量 (原子操作)
        # 4. 旧版本标记为 standby

    def _deploy_rolling(self, config: ReloadConfig, version: StrategyVersion, event: ReloadEvent):
        """滚动更新：逐个实例更新."""
        logger.info(f"Rolling deploy {config.strategy_id} -> {version.version_id}")
        # 实际由容器编排器处理

    def _deploy_canary(self, config: ReloadConfig, version: StrategyVersion, event: ReloadEvent):
        """灰度发布：小比例流量先行."""
        logger.info(f"Canary deploy {config.strategy_id} -> {version.version_id} ({config.canary_percentage}%)")
        version.status = StrategyVersionStatus.STANDBY
        # 1. 启动少量实例
        # 2. 监控指标
        # 3. 渐进扩大或回滚

    def _deploy_recreate(self, config: ReloadConfig, version: StrategyVersion, event: ReloadEvent):
        """重建：停止旧版本，启动新版本."""
        logger.info(f"Recreate deploy {config.strategy_id} -> {version.version_id}")

    def _wait_healthy(self, config: ReloadConfig, version: StrategyVersion):
        """等待健康检查通过."""
        if not config.health_check:
            return
        start = time.time()
        while time.time() - start < config.health_check_timeout:
            if config.health_check():
                return
            time.sleep(config.health_check_interval)
        raise RuntimeError("Health check timeout")

    def rollback(self, strategy_id: str, target_version: Optional[str] = None) -> ReloadEvent:
        """回滚到指定版本."""
        versions = self._versions.get(strategy_id, [])
        if not versions:
            raise ValueError(f"No versions for {strategy_id}")

        if target_version:
            version = next((v for v in versions if v.version_id == target_version), None)
        else:
            # 回滚到前一个 ACTIVE 版本
            active = self._active_version.get(strategy_id)
            idx = next((i for i, v in enumerate(versions) if v.version_id == active), -1)
            version = versions[idx - 1] if idx > 0 else None

        if not version:
            raise ValueError("No version to rollback to")

        event = self.trigger_reload(strategy_id, ReloadTrigger.API, version.version_id)
        version.rolled_back_at = datetime.utcnow()
        version.status = StrategyVersionStatus.ROLLED_BACK
        return event

    def on_reload(self, strategy_id: str, callback: Callable[[ReloadEvent], None]):
        """注册重载回调."""
        self._callbacks[strategy_id].append(callback)

    def get_history(self, strategy_id: str, limit: int = 50) -> List[ReloadEvent]:
        with self._lock:
            events = [e for e in self._event_history if e.strategy_id == strategy_id]
            return events[-limit:]

    def get_active_version(self, strategy_id: str) -> Optional[StrategyVersion]:
        version_id = self._active_version.get(strategy_id)
        if not version_id:
            return None
        for v in self._versions.get(strategy_id, []):
            if v.version_id == version_id:
                return v
        return None


# 全局实例
_hot_reloader: Optional[StrategyHotReloader] = None


def get_hot_reloader() -> StrategyHotReloader:
    global _hot_reloader
    if _hot_reloader is None:
        _hot_reloader = StrategyHotReloader()
    return _hot_reloader


def init_hot_reloader() -> StrategyHotReloader:
    global _hot_reloader
    _hot_reloader = StrategyHotReloader()
    return _hot_reloader
