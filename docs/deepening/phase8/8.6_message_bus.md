"""策略间通信总线 - Redis Streams/Kafka、Schema Registry、DLQ."""

from __future__ import annotations
import json
import logging
import threading
import time
import uuid
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any, Callable, Dict, List, Optional, Type, TypeVar

from quant.utils.logger import get_logger
from quant.config.constants import _require_cfg

logger = get_logger("tenant.message_bus")

T = TypeVar("T")


class TransportType(Enum):
    """传输层类型."""
    REDIS_STREAMS = "redis_streams"
    KAFKA = "kafka"
    IN_MEMORY = "in_memory"  # 测试/开发


class MessagePriority(Enum):
    """消息优先级."""
    LOW = 0
    NORMAL = 50
    HIGH = 100
    CRITICAL = 200


@dataclass
class MessageSchema:
    """消息 Schema 定义."""
    name: str
    version: int
    fields: Dict[str, str]  # field_name -> type (json schema 简化)
    required: List[str] = field(default_factory=list)
    description: str = ""

    def validate(self, data: Dict) -> Tuple[bool, List[str]]:
        """验证消息数据."""
        errors = []
        for field_name in self.required:
            if field_name not in data:
                errors.append(f"Missing required field: {field_name}")
        for field_name, field_type in self.fields.items():
            if field_name in data:
                if not self._check_type(data[field_name], field_type):
                    errors.append(f"Field {field_name} type mismatch: expected {field_type}")
        return len(errors) == 0, errors

    def _check_type(self, value: Any, expected: str) -> bool:
        type_map = {
            "string": str, "integer": int, "number": (int, float),
            "boolean": bool, "array": list, "object": dict,
        }
        expected_type = type_map.get(expected.lower())
        if expected_type:
            return isinstance(value, expected_type)
        return True


class SchemaRegistry:
    """Schema 注册表 - 支持版本演进."""

    def __init__(self):
        self._schemas: Dict[str, Dict[int, MessageSchema]] = {}  # name -> {version -> schema}
        self._latest_version: Dict[str, int] = {}
        self._lock = threading.RLock()

    def register(self, schema: MessageSchema) -> int:
        """注册 Schema，返回版本号."""
        with self._lock:
            if schema.name not in self._schemas:
                self._schemas[schema.name] = {}
            version = schema.version
            if version in self._schemas[schema.name]:
                # 检查兼容性
                if not self._is_compatible(self._schemas[schema.name][version], schema):
                    raise ValueError(f"Schema {schema.name} v{version} is not backward compatible")
            self._schemas[schema.name][version] = schema
            self._latest_version[schema.name] = max(self._latest_version.get(schema.name, 0), version)
            logger.info(f"Registered schema: {schema.name} v{version}")
            return version

    def get(self, name: str, version: Optional[int] = None) -> Optional[MessageSchema]:
        with self._lock:
            versions = self._schemas.get(name, {})
            if not versions:
                return None
            if version is None:
                version = self._latest_version.get(name, 1)
            return versions.get(version)

    def get_latest(self, name: str) -> Optional[MessageSchema]:
        return self.get(name)

    def _is_compatible(self, old: MessageSchema, new: MessageSchema) -> bool:
        """检查向后兼容性：新增字段必须可选，不能删除必填字段."""
        # 不能删除 required 字段
        for field in old.required:
            if field not in new.fields:
                return False
        # 类型不能变更（除非是扩展）
        for field, old_type in old.fields.items():
            if field in new.fields and new.fields[field] != old_type:
                return False
        return True

    def list_schemas(self) -> List[str]:
        with self._lock:
            return list(self._schemas.keys())


@dataclass
class Message:
    """通用消息."""
    topic: str
    payload: Dict[str, Any]
    schema_name: str = ""
    schema_version: int = 1
    message_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    correlation_id: str = ""  # 用于链路追踪
    timestamp: datetime = field(default_factory=datetime.utcnow)
    priority: MessagePriority = MessagePriority.NORMAL
    headers: Dict[str, str] = field(default_factory=dict)
    retry_count: int = 0
    max_retries: int = 3

    def to_json(self) -> str:
        return json.dumps({
            "topic": self.topic,
            "payload": self.payload,
            "schema_name": self.schema_name,
            "schema_version": self.schema_version,
            "message_id": self.message_id,
            "correlation_id": self.correlation_id,
            "timestamp": self.timestamp.isoformat(),
            "priority": self.priority.value,
            "headers": self.headers,
            "retry_count": self.retry_count,
            "max_retries": self.max_retries,
        })

    @classmethod
    def from_json(cls, data: str) -> "Message":
        obj = json.loads(data)
        obj["timestamp"] = datetime.fromisoformat(obj["timestamp"])
        obj["priority"] = MessagePriority(obj["priority"])
        return cls(**obj)


class DeadLetterQueue:
    """死信队列."""

    def __init__(self, max_size: int = 10000):
        self._queue: List[Tuple[Message, str]] = []  # (message, error)
        self._max_size = max_size
        self._lock = threading.RLock()

    def add(self, message: Message, error: str):
        with self._lock:
            if len(self._queue) >= self._max_size:
                self._queue.pop(0)  # 丢弃最旧
            self._queue.append((message, error))

    def get_all(self) -> List[Tuple[Message, str]]:
        with self._lock:
            return list(self._queue)

    def retry(self, index: int) -> Optional[Message]:
        with self._lock:
            if 0 <= index < len(self._queue):
                msg, _ = self._queue.pop(index)
                msg.retry_count = 0  # 重置重试
                return msg
        return None

    def clear(self):
        with self._lock:
            self._queue.clear()

    def size(self) -> int:
        with self._lock:
            return len(self._queue)


class Transport(ABC):
    """传输层抽象."""

    @abstractmethod
    def publish(self, topic: str, message: Message) -> bool:
        pass

    @abstractmethod
    def subscribe(self, topic: str, group: str, handler: Callable[[Message], None]) -> str:
        pass

    @abstractmethod
    def unsubscribe(self, subscription_id: str) -> bool:
        pass

    @abstractmethod
    def close(self):
        pass


class RedisStreamsTransport(Transport):
    """Redis Streams 传输层."""

    def __init__(self, redis_url: str = "redis://localhost:6379"):
        import redis
        self._client = redis.from_url(redis_url, decode_responses=True)
        self._subscriptions: Dict[str, Dict] = {}
        self._consumer_threads: Dict[str, threading.Thread] = {}
        self._running = False
        self._lock = threading.RLock()

    def publish(self, topic: str, message: Message) -> bool:
        try:
            # 使用 XADD 添加到流
            stream_key = f"quant:{topic}"
            msg_data = {
                "data": message.to_json(),
            }
            self._client.xadd(stream_key, msg_data, maxlen=100000)
            return True
        except Exception as e:
            logger.error(f"Redis publish failed: {e}")
            return False

    def subscribe(self, topic: str, group: str, handler: Callable[[Message], None]) -> str:
        sub_id = str(uuid.uuid4())[:8]
        stream_key = f"quant:{topic}"

        # 创建消费者组
        try:
            self._client.xgroup_create(stream_key, group, id="0", mkstream=True)
        except Exception:
            pass  # 组已存在

        def consumer_loop():
            consumer_name = f"{group}-{sub_id}"
            while self._running:
                try:
                    # 阻塞读取
                    result = self._client.xreadgroup(
                        group, consumer_name, {stream_key: ">"}, count=10, block=5000
                    )
                    for stream, messages in result:
                        for msg_id, msg_data in messages:
                            try:
                                message = Message.from_json(msg_data["data"])
                                handler(message)
                                self._client.xack(stream_key, group, msg_id)
                            except Exception as e:
                                logger.error(f"Message handler error: {e}")
                                # 不 ack，重新投递
                except Exception as e:
                    logger.error(f"Consumer error: {e}")
                    time.sleep(1)

        thread = threading.Thread(target=consumer_loop, daemon=True, name=f"redis-consumer-{sub_id}")
        thread.start()

        with self._lock:
            self._subscriptions[sub_id] = {"topic": topic, "group": group, "thread": thread}

        return sub_id

    def unsubscribe(self, subscription_id: str) -> bool:
        with self._lock:
            sub = self._subscriptions.pop(subscription_id, None)
            if sub:
                # 线程会在下次循环退出
                return True
        return False

    def close(self):
        self._running = False
        for sub in self._subscriptions.values():
            if sub["thread"].is_alive():
                sub["thread"].join(timeout=5)


class InMemoryTransport(Transport):
    """内存传输层 (测试/开发用)."""

    def __init__(self):
        self._topics: Dict[str, List[Message]] = {}
        self._subscribers: Dict[str, List[Callable[[Message], None]]] = {}
        self._lock = threading.RLock()

    def publish(self, topic: str, message: Message) -> bool:
        with self._lock:
            if topic not in self._topics:
                self._topics[topic] = []
            self._topics[topic].append(message)
            # 同步分发
            for handler in self._subscribers.get(topic, []):
                try:
                    handler(message)
                except Exception as e:
                    logger.error(f"In-memory handler error: {e}")
        return True

    def subscribe(self, topic: str, group: str, handler: Callable[[Message], None]) -> str:
        with self._lock:
            if topic not in self._subscribers:
                self._subscribers[topic] = []
            self._subscribers[topic].append(handler)
        return str(uuid.uuid4())[:8]

    def unsubscribe(self, subscription_id: str) -> bool:
        return True

    def close(self):
        pass


class MessageBus:
    """消息总线 - 统一入口."""

    def __init__(
        self,
        transport: TransportType = TransportType.IN_MEMORY,
        redis_url: str = "redis://localhost:6379",
        schema_registry: Optional[SchemaRegistry] = None,
    ):
        self.transport_type = transport
        self._transport = self._create_transport(transport, redis_url)
        self._schema_registry = schema_registry or SchemaRegistry()
        self._dlq = DeadLetterQueue()
        self._handlers: Dict[str, List[Callable[[Message], None]]] = {}
        self._topic_schemas: Dict[str, Tuple[str, int]] = {}  # topic -> (schema_name, version)
        self._lock = threading.RLock()

    def _create_transport(self, transport: TransportType, redis_url: str) -> Transport:
        if transport == TransportType.REDIS_STREAMS:
            return RedisStreamsTransport(redis_url)
        elif transport == TransportType.KAFKA:
            # return KafkaTransport(bootstrap_servers)
            raise NotImplementedError("Kafka transport not yet implemented")
        return InMemoryTransport()

    def register_schema(self, schema: MessageSchema):
        """注册消息 Schema."""
        self._schema_registry.register(schema)

    def bind_schema(self, topic: str, schema_name: str, version: int = 1):
        """绑定 Topic 到 Schema."""
        with self._lock:
            self._topic_schemas[topic] = (schema_name, version)

    def publish(self, topic: str, payload: Dict, **kwargs) -> bool:
        """发布消息."""
        schema_name, schema_version = self._topic_schemas.get(topic, ("", 1))

        # Schema 验证
        if schema_name:
            schema = self._schema_registry.get(schema_name, schema_version)
            if schema:
                valid, errors = schema.validate(payload)
                if not valid:
                    logger.error(f"Schema validation failed for {topic}: {errors}")
                    return False

        message = Message(
            topic=topic,
            payload=payload,
            schema_name=schema_name,
            schema_version=schema_version,
            **kwargs,
        )

        return self._transport.publish(topic, message)

    def subscribe(
        self,
        topic: str,
        group: str,
        handler: Callable[[Message], None],
        auto_retry: bool = True,
    ) -> str:
        """订阅消息."""
        def wrapped_handler(message: Message):
            try:
                # Schema 验证
                if message.schema_name:
                    schema = self._schema_registry.get(message.schema_name, message.schema_version)
                    if schema:
                        valid, errors = schema.validate(message.payload)
                        if not valid:
                            logger.error(f"Received invalid message: {errors}")
                            if auto_retry and message.retry_count < message.max_retries:
                                message.retry_count += 1
                                self._transport.publish(message.topic, message)
                            else:
                                self._dlq.add(message, f"Schema validation failed: {errors}")
                            return

                handler(message)
            except Exception as e:
                logger.error(f"Handler error for {topic}: {e}")
                if auto_retry and message.retry_count < message.max_retries:
                    message.retry_count += 1
                    self._transport.publish(message.topic, message)
                else:
                    self._dlq.add(message, str(e))

        return self._transport.subscribe(topic, group, wrapped_handler)

    def unsubscribe(self, subscription_id: str) -> bool:
        return self._transport.unsubscribe(subscription_id)

    def get_dlq(self) -> DeadLetterQueue:
        return self._dlq

    def close(self):
        self._transport.close()


# 全局实例
_message_bus: Optional[MessageBus] = None


def get_message_bus() -> MessageBus:
    global _message_bus
    if _message_bus is None:
        _message_bus = MessageBus()
    return _message_bus


def init_message_bus(**kwargs) -> MessageBus:
    global _message_bus
    _message_bus = MessageBus(**kwargs)
    return _message_bus

signal_schema = MessageSchema(

    version=1,
    fields={
        "symbol": "string",
        "signal": "string",  # BUY/SELL/HOLD
        "strength": "number",
        "strategy_id": "string",
        "timestamp": "integer",
    },
    required=["symbol", "signal", "strategy_id"],

order_schema = MessageSchema(
    name="order_request",
    version=1,
    fields={
        "order_id": "string",
        "symbol": "string",
        "side": "string",  # BUY/SELL
        "quantity": "number",
        "price": "number",
        "order_type": "string",  # MARKET/LIMIT
        "strategy_id": "string",
    },
    required=["order_id", "symbol", "side", "quantity", "strategy_id"],
)

# 注册
bus = init_message_bus(transport=TransportType.REDIS_STREAMS)
bus.register_schema(signal_schema)
bus.register_schema(order_schema)

# 绑定 Topic
bus.bind_schema("signals", "trading_signal", 1)
bus.bind_schema("orders", "order_request", 1)

# 发布
bus.publish("signals", {
    "symbol": "000001.SZ",
    "signal": "BUY",
    "strength": 0.85,
    "strategy_id": "momentum_v1",
    "timestamp": int(time.time() * 1000),
})

# 订阅
def on_signal(msg: Message):
    print(f"Received signal: {msg.payload}")

bus.subscribe("signals", "risk_engine_group", on_signal)
