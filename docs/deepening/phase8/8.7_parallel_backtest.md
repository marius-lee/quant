# quant/tenant/parallel_backtest.py
"""多策略并行回测 - 隔离并行、资源配额、结果聚合."""

from __future__ import annotations
import concurrent.futures
import hashlib
import json
import logging
import threading
import time
import uuid
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from pathlib import Path
from typing import Any, Callable, Dict, List, Optional, Set

from quant.utils.logger import get_logger
from quant.config.constants import _require_cfg

logger = get_logger("tenant.parallel_backtest")


class BacktestStatus(Enum):
    """回测状态."""
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"


@dataclass
class BacktestConfig:
    """单策略回测配置."""
    strategy_id: str
    strategy_class: str           # 策略类路径
    config: Dict[str, Any]        # 策略参数
    start_date: str
    end_date: str
    initial_capital: float = 5000.0
    benchmark: str = "000300.SH"
    data_frequency: str = "1d"
    output_dir: Optional[str] = None


@dataclass
class BacktestResult:
    """回测结果."""
    strategy_id: str
    config_hash: str
    status: BacktestStatus = BacktestStatus.PENDING
    metrics: Dict[str, float] = field(default_factory=dict)  # sharpe, max_dd, return, etc.
    trades: List[Dict] = field(default_factory=list)
    equity_curve: List[float] = field(default_factory=list)
    daily_returns: List[float] = field(default_factory=list)
    error: Optional[str] = None
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    duration_seconds: float = 0.0
    output_path: Optional[str] = None


@dataclass
class ParallelBacktestJob:
    """并行回测作业."""
    job_id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    name: str = ""
    configs: List[BacktestConfig] = field(default_factory=list)
    max_workers: int = 4
    resource_limits: Dict[str, Any] = field(default_factory=dict)  # cpu, memory per worker
    status: BacktestStatus = BacktestStatus.PENDING
    created_at: datetime = field(default_factory=datetime.utcnow)
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    results: List[BacktestResult] = field(default_factory=list)
    aggregated_report: Optional[Dict] = None


class ResourcePool:
    """资源池 - 控制并行度."""

    def __init__(self, max_cpu: float = 8.0, max_memory_gb: float = 16.0):
        self.max_cpu = max_cpu
        self.max_memory_gb = max_memory_gb
        self.allocated_cpu = 0.0
        self.allocated_memory = 0.0
        self._lock = threading.Lock()
        self._waiters: List[threading.Event] = []

    def acquire(self, cpu: float, memory_gb: float) -> bool:
        with self._lock:
            if self.allocated_cpu + cpu <= self.max_cpu and self.allocated_memory + memory_gb <= self.max_memory_gb:
                self.allocated_cpu += cpu
                self.allocated_memory += memory_gb
                return True
            return False

    def release(self, cpu: float, memory_gb: float):
        with self._lock:
            self.allocated_cpu = max(0, self.allocated_cpu - cpu)
            self.allocated_memory = max(0, self.allocated_memory - memory_gb)
            # 唤醒等待者
            for event in self._waiters:
                event.set()
            self._waiters.clear()

    def wait_for_resources(self, cpu: float, memory_gb: float, timeout: float = 300.0) -> bool:
        """等待资源可用."""
        start = time.time()
        while time.time() - start < timeout:
            if self.acquire(cpu, memory_gb):
                return True
            event = threading.Event()
            with self._lock:
                self._waiters.append(event)
            event.wait(timeout=5.0)
        return False


class StrategyExecutor:
    """策略执行器 - 隔离运行单个回测."""

    def __init__(self, resource_pool: ResourcePool):
        self.resource_pool = resource_pool
        self._running: Dict[str, threading.Thread] = {}

    def execute(self, config: BacktestConfig, cpu: float = 1.0, memory_gb: float = 2.0) -> BacktestResult:
        """执行单策略回测."""
        result = BacktestResult(
            strategy_id=config.strategy_id,
            config_hash=self._compute_config_hash(config),
        )

        # 等待资源
        if not self.resource_pool.wait_for_resources(cpu, memory_gb):
            result.status = BacktestStatus.FAILED
            result.error = "Resource acquisition timeout"
            return result

        result.status = BacktestStatus.RUNNING
        result.started_at = datetime.utcnow()
        start_time = time.time()

        try:
            # 动态导入策略
            strategy_module = self._import_strategy(config.strategy_class)
            strategy_instance = strategy_module.Strategy(**config.config)

            # 运行回测 (这里调用实际的回测引擎)
            from quant.backtest.loop import run_backtest

            backtest_result = run_backtest(
                start_date=config.start_date,
                end_date=config.end_date,
                capital=config.initial_capital,
                strategy=strategy_instance,
            )

            result.metrics = backtest_result.get("metrics", {})
            result.trades = backtest_result.get("trades", [])
            result.equity_curve = backtest_result.get("equity_curve", [])
            result.daily_returns = backtest_result.get("daily_returns", [])
            result.status = BacktestStatus.COMPLETED

        except Exception as e:
            result.status = BacktestStatus.FAILED
            result.error = str(e)
            logger.error(f"Backtest failed for {config.strategy_id}: {e}")

        finally:
            result.completed_at = datetime.utcnow()
            result.duration_seconds = time.time() - start_time
            self.resource_pool.release(cpu, memory_gb)

        return result

    def _import_strategy(self, class_path: str):
        """动态导入策略类."""
        module_path, class_name = class_path.rsplit(".", 1)
        module = __import__(module_path, fromlist=[class_name])
        return getattr(module, class_name)

    def _compute_config_hash(self, config: BacktestConfig) -> str:
        """计算配置哈希，用于增量回测判断."""
        data = {
            "strategy_class": config.strategy_class,
            "config": config.config,
            "start_date": config.start_date,
            "end_date": config.end_date,
            "capital": config.initial_capital,
        }
        return hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()[:16]


class ParallelBacktestEngine:
    """并行回测引擎."""

    def __init__(self, max_workers: int = 4, max_cpu: float = 8.0, max_memory_gb: float = 16.0):
        self.max_workers = max_workers
        self.resource_pool = ResourcePool(max_cpu, max_memory_gb)
        self.executor = StrategyExecutor(self.resource_pool)
        self._jobs: Dict[str, ParallelBacktestJob] = {}
        self._executor_pool: Optional[concurrent.futures.ThreadPoolExecutor] = None
        self._lock = threading.RLock()

    def submit_job(self, job: ParallelBacktestJob) -> str:
        """提交并行回测作业."""
        with self._lock:
            self._jobs[job.job_id] = job
        return job.job_id

    def run_job(self, job_id: str) -> ParallelBacktestJob:
        """运行作业 (阻塞直到完成)."""
        with self._lock:
            job = self._jobs.get(job_id)
            if not job:
                raise ValueError(f"Job {job_id} not found")

        job.status = BacktestStatus.RUNNING
        job.started_at = datetime.utcnow()

        # 创建线程池
        self._executor_pool = concurrent.futures.ThreadPoolExecutor(max_workers=job.max_workers)

        try:
            # 提交所有任务
            future_to_config = {}
            for config in job.configs:
                future = self._executor_pool.submit(
                    self.executor.execute,
                    config,
                    job.resource_limits.get("cpu_per_worker", 1.0),
                    job.resource_limits.get("memory_gb_per_worker", 2.0),
                )
                future_to_config[future] = config

            # 收集结果
            for future in concurrent.futures.as_completed(future_to_config):
                config = future_to_config[future]
                try:
                    result = future.result()
                    job.results.append(result)
                except Exception as e:
                    result = BacktestResult(
                        strategy_id=config.strategy_id,
                        config_hash=self.executor._compute_config_hash(config),
                        status=BacktestStatus.FAILED,
                        error=str(e),
                    )
                    job.results.append(result)

            # 聚合报告
            job.aggregated_report = self._aggregate_results(job.results)

            # 判断整体状态
            failed = sum(1 for r in job.results if r.status == BacktestStatus.FAILED)
            if failed == len(job.results):
                job.status = BacktestStatus.FAILED
            elif failed > 0:
                job.status = BacktestStatus.COMPLETED  # 部分成功
            else:
                job.status = BacktestStatus.COMPLETED

        finally:
            if self._executor_pool:
                self._executor_pool.shutdown(wait=True)
                self._executor_pool = None
            job.completed_at = datetime.utcnow()

        return job

    def _aggregate_results(self, results: List[BacktestResult]) -> Dict:
        """聚合多策略回测结果."""
        successful = [r for r in results if r.status == BacktestStatus.COMPLETED]

        if not successful:
            return {"error": "All strategies failed"}

        # 计算统计指标
        sharpe_ratios = [r.metrics.get("sharpe_ratio", 0) for r in successful]
        max_drawdowns = [r.metrics.get("max_drawdown", 0) for r in successful]
        returns = [r.metrics.get("total_return", 0) for r in successful]
        win_rates = [r.metrics.get("win_rate", 0) for r in successful]

        return {
            "total_strategies": len(results),
            "successful": len(successful),
            "failed": len(results) - len(successful),
            "avg_sharpe": sum(sharpe_ratios) / len(sharpe_ratios),
            "best_sharpe": max(sharpe_ratios),
            "worst_sharpe": min(sharpe_ratios),
            "avg_max_drawdown": sum(max_drawdowns) / len(max_drawdowns),
            "worst_max_drawdown": max(max_drawdowns),
            "avg_return": sum(returns) / len(returns),
            "best_return": max(returns),
            "avg_win_rate": sum(win_rates) / len(win_rates),
            "top_strategies": sorted(
                [{"id": r.strategy_id, "sharpe": r.metrics.get("sharpe_ratio", 0)} for r in successful],
                key=lambda x: x["sharpe"],
                reverse=True
            )[:5],
        }

    def cancel_job(self, job_id: str) -> bool:
        """取消作业."""
        with self._lock:
            job = self._jobs.get(job_id)
            if not job or job.status not in (BacktestStatus.PENDING, BacktestStatus.RUNNING):
                return False
            job.status = BacktestStatus.CANCELLED
            # 实际取消需要更复杂的机制
            return True

    def get_job(self, job_id: str) -> Optional[ParallelBacktestJob]:
        with self._lock:
            return self._jobs.get(job_id)

    def list_jobs(self) -> List[ParallelBacktestJob]:
        with self._lock:
            return list(self._jobs.values())


class IncrementalBacktestManager:
    """增量回测管理器 - 仅重算变更的策略/日期."""

    def __init__(self, engine: ParallelBacktestEngine, cache_dir: Path):
        self.engine = engine
        self.cache_dir = cache_dir
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self._result_cache: Dict[str, BacktestResult] = {}

    def load_cache(self):
        """加载缓存."""
        cache_file = self.cache_dir / "backtest_cache.json"
        if cache_file.exists():
            with open(cache_file) as f:
                data = json.load(f)
                # 反序列化
                pass

    def save_cache(self):
        """保存缓存."""
        cache_file = self.cache_dir / "backtest_cache.json"
        # 序列化
        pass

    def should_skip(self, config: BacktestConfig) -> Optional[BacktestResult]:
        """检查是否可以跳过 (配置未变且有缓存)."""
        config_hash = self.engine.executor._compute_config_hash(config)
        cached = self._result_cache.get(config_hash)
        if cached and cached.status == BacktestStatus.COMPLETED:
            logger.info(f"Cache hit for {config.strategy_id}: {config_hash}")
            return cached
        return None

    def run_incremental(self, job: ParallelBacktestJob) -> ParallelBacktestJob:
        """运行增量回测：跳过未变更的策略."""
        # 分离已缓存和需重算
        to_run = []
        cached_results = []

        for config in job.configs:
            cached = self.should_skip(config)
            if cached:
                cached_results.append(cached)
            else:
                to_run.append(config)

        logger.info(f"Incremental backtest: {len(cached_results)} cached, {len(to_run)} to run")

        # 创建新作业只运行需重算的
        if to_run:
            new_job = ParallelBacktestJob(
                name=f"{job.name}_incremental",
                configs=to_run,
                max_workers=job.max_workers,
                resource_limits=job.resource_limits,
            )
            self.engine.run_job(new_job.job_id)
            job.results = cached_results + new_job.results
        else:
            job.results = cached_results

        job.aggregated_report = self.engine._aggregate_results(job.results)
        return job

def get_parallel_backtest_engine() -> ParallelBacktestEngine:
    global _parallel_backtest_engine
    if _parallel_backtest_engine is None:
        _parallel_backtest_engine = ParallelBacktestEngine()
    return _parallel_backtest_engine


def init_parallel_backtest_engine(**kwargs) -> ParallelBacktestEngine:
    global _parallel_backtest_engine
    _parallel_backtest_engine = ParallelBacktestEngine(**kwargs)
    return _parallel_backtest_engine
