### 1. 应用层健康检查 (`quant/web/health.py`)

```python
# quant/web/health.py
"""健康检查端点 - Liveness/Readiness/Startup."""

from __future__ import annotations
import time
import psutil
from dataclasses import dataclass
from datetime import datetime
from enum import Enum
from typing import Dict, Any, Optional
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy import text

from quant.utils.logger import get_logger
from quant.data.store import DataStore
from quant.config.constants import _require_cfg

logger = get_logger("web.health")

router = APIRouter(prefix="/health", tags=["health"])

class HealthStatus(str, Enum):
    HEALTHY = "healthy"
    DEGRADED = "degraded"
    UNHEALTHY = "unhealthy"

@dataclass
class DependencyCheck:
    name: str
    status: HealthStatus
    latency_ms: float
    message: str = ""
    details: Dict[str, Any] = None

@dataclass
class HealthResponse:
    status: HealthStatus
    timestamp: datetime
    version: str
    uptime_seconds: float
    checks: Dict[str, DependencyCheck]
    metadata: Dict[str, Any] = None

# 应用启动时间
_start_time = time.time()

def get_uptime() -> float:
    return time.time() - _start_time

async def check_database() -> DependencyCheck:
    """检查 PostgreSQL 连接."""
    start = time.time()
    try:
        store = DataStore()
        with store.get_connection() as conn:
            conn.execute(text("SELECT 1"))
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="postgresql",
            status=HealthStatus.HEALTHY,
            latency_ms=latency,
            message="Connection OK"
        )
    except Exception as e:
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="postgresql",
            status=HealthStatus.UNHEALTHY,
            latency_ms=latency,
            message=str(e)
        )

async def check_redis() -> DependencyCheck:
    """检查 Redis 连接."""
    start = time.time()
    try:
        import redis
        client = redis.from_url(_require_cfg("redis.url", "redis://localhost:6379"))
        client.ping()
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="redis",
            status=HealthStatus.HEALTHY,
            latency_ms=latency,
            message="PONG"
        )
    except Exception as e:
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="redis",
            status=HealthStatus.UNHEALTHY,
            latency_ms=latency,
            message=str(e)
        )

async def check_minio() -> DependencyCheck:
    """检查 MinIO 连接."""
    start = time.time()
    try:
        from minio import Minio
        client = Minio(
            _require_cfg("minio.endpoint", "localhost:9000"),
            access_key=_require_cfg("minio.access_key"),
            secret_key=_require_cfg("minio.secret_key"),
            secure=False
        )
        client.list_buckets()
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="minio",
            status=HealthStatus.HEALTHY,
            latency_ms=latency,
            message="Buckets accessible"
        )
    except Exception as e:
        latency = (time.time() - start) * 1000
        return DependencyCheck(
            name="minio",
            status=HealthStatus.DEGRADED,
            latency_ms=latency,
            message=str(e)
        )

async def check_disk_space() -> DependencyCheck:
    """检查磁盘空间."""
    try:
        usage = psutil.disk_usage('/')
        free_percent = (usage.free / usage.total) * 100
        status = HealthStatus.HEALTHY
        if free_percent < 10:
            status = HealthStatus.UNHEALTHY
        elif free_percent < 20:
            status = HealthStatus.DEGRADED
        return DependencyCheck(
            name="disk",
            status=status,
            latency_ms=0,
            message=f"Free: {free_percent:.1f}%",
            details={"total_gb": usage.total / 1e9, "free_gb": usage.free / 1e9}
        )
    except Exception as e:
        return DependencyCheck(
            name="disk",
            status=HealthStatus.UNHEALTHY,
            latency_ms=0,
            message=str(e)
        )

async def check_memory() -> DependencyCheck:
    """检查内存使用."""
    try:
        mem = psutil.virtual_memory()
        status = HealthStatus.HEALTHY
        if mem.percent > 90:
            status = HealthStatus.UNHEALTHY
        elif mem.percent > 80:
            status = HealthStatus.DEGRADED
        return DependencyCheck(
            name="memory",
            status=status,
            latency_ms=0,
            message=f"Usage: {mem.percent:.1f}%",
            details={"total_gb": mem.total / 1e9, "available_gb": mem.available / 1e9}
        )
    except Exception as e:
        return DependencyCheck(
            name="memory",
            status=HealthStatus.UNHEALTHY,
            latency_ms=0,
            message=str(e)
        )

@router.get("/live", summary="Liveness Probe - 进程是否存活")
async def liveness():
    """Kubernetes Liveness Probe - 仅检查进程是否运行."""
    return {
        "status": "alive",
        "timestamp": datetime.utcnow().isoformat(),
        "uptime_seconds": get_uptime()
    }

@router.get("/ready", summary="Readiness Probe - 是否可服务流量")
async def readiness():
    """Kubernetes Readiness Probe - 检查所有依赖是否就绪."""
    checks = await run_all_checks()
    
    # 任何关键依赖不健康 -> not ready
    unhealthy = [c for c in checks.values() if c.status == HealthStatus.UNHEALTHY]
    degraded = [c for c in checks.values() if c.status == HealthStatus.DEGRADED]
    
    if unhealthy:
        overall = HealthStatus.UNHEALTHY
    elif degraded:
        overall = HealthStatus.DEGRADED
    else:
        overall = HealthStatus.HEALTHY
    
    response = HealthResponse(
        status=overall,
        timestamp=datetime.utcnow(),
        version=_require_cfg("app.version", "1.0.0"),
        uptime_seconds=get_uptime(),
        checks=checks
    )
    
    http_status = status.HTTP_200_OK if overall != HealthStatus.UNHEALTHY else status.HTTP_503_SERVICE_UNAVAILABLE
    return response.__dict__, http_status

@router.get("/startup", summary="Startup Probe - 启动完成检查")
async def startup():
    """Kubernetes Startup Probe - 检查应用是否完成初始化."""
    checks = await run_all_checks()
    
    # 启动阶段允许降级，但关键依赖必须健康
    critical_unhealthy = [
        c for c in checks.values() 
        if c.status == HealthStatus.UNHEALTHY and c.name in ("postgresql", "redis")
    ]
    
    if critical_unhealthy:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail={"status": "starting", "checks": {k: v.__dict__ for k, v in checks.items()}}
        )
    
    return {
        "status": "started",
        "timestamp": datetime.utcnow().isoformat(),
        "uptime_seconds": get_uptime()
    }

@router.get("/", summary="完整健康检查")
async def health():
    """完整健康检查 - 返回详细信息."""
    checks = await run_all_checks()
    
    unhealthy = [c for c in checks.values() if c.status == HealthStatus.UNHEALTHY]
    degraded = [c for c in checks.values() if c.status == HealthStatus.DEGRADED]
    
    if unhealthy:
        overall = HealthStatus.UNHEALTHY
    elif degraded:
        overall = HealthStatus.DEGRADED
    else:
        overall = HealthStatus.HEALTHY
    
    response = HealthResponse(
        status=overall,
        timestamp=datetime.utcnow(),
        version=_require_cfg("app.version", "1.0.0"),
        uptime_seconds=get_uptime(),
        checks=checks,
        metadata={
            "environment": _require_cfg("app.environment", "production"),
            "python_version": "3.12",
        }
    )
    
    return response.__dict__

async def run_all_checks() -> Dict[str, DependencyCheck]:
    """并行运行所有依赖检查."""
    import asyncio
    results = await asyncio.gather(
        check_database(),
        check_redis(),
        check_minio(),
        check_disk_space(),
        check_memory(),
        return_exceptions=True
    )
    
    checks = {}
    for r in results:
        if isinstance(r, Exception):
            logger.error(f"Health check failed: {r}")
            checks["unknown"] = DependencyCheck(
                name="unknown",
                status=HealthStatus.UNHEALTHY,
                latency_ms=0,
                message=str(r)
            )
        else:
            checks[r.name] = r
    return checks
```

### 2. 优雅关闭处理 (`quant/web/lifecycle.py`)

```python
# quant/web/lifecycle.py
"""应用生命周期管理 - 优雅启动/关闭."""

from __future__ import annotations
import asyncio
import signal
import sys
from contextlib import asynccontextmanager
from typing import Callable, List

from fastapi import FastAPI
from quant.utils.logger import get_logger

logger = get_logger("web.lifecycle")

class LifecycleManager:
    """生命周期管理器."""
    
    def __init__(self):
        self._startup_hooks: List[Callable] = []
        self._shutdown_hooks: List[Callable] = []
        self._shutdown_event = asyncio.Event()
        self._is_shutting_down = False
    
    def on_startup(self, hook: Callable):
        """注册启动钩子."""
        self._startup_hooks.append(hook)
    
    def on_shutdown(self, hook: Callable):
        """注册关闭钩子."""
        self._shutdown_hooks.append(hook)
    
    async def startup(self):
        """执行启动钩子."""
        logger.info("Running startup hooks...")
        for hook in self._startup_hooks:
            try:
                if asyncio.iscoroutinefunction(hook):
                    await hook()
                else:
                    hook()
            except Exception as e:
                logger.error(f"Startup hook failed: {e}")
                raise
        logger.info("All startup hooks completed")
    
    async def shutdown(self):
        """执行关闭钩子."""
        if self._is_shutting_down:
            return
        self._is_shutting_down = True
        logger.info("Running shutdown hooks...")
        
        # 并行执行关闭钩子，但设置超时
        tasks = []
        for hook in self._shutdown_hooks:
            if asyncio.iscoroutinefunction(hook):
                tasks.append(asyncio.create_task(hook()))
            else:
                tasks.append(asyncio.create_task(asyncio.to_thread(hook)))
        
        if tasks:
            done, pending = await asyncio.wait(
                tasks,
                timeout=30.0,
                return_when=asyncio.ALL_COMPLETED
            )
            
            for task in pending:
                task.cancel()
                try:
                    await task
                except asyncio.CancelledError:
                    pass
        
        self._shutdown_event.set()
        logger.info("All shutdown hooks completed")
    
    async def wait_for_shutdown(self):
        """等待关闭信号."""
        await self._shutdown_event.wait()

# 全局实例
_lifecycle_manager = LifecycleManager()

def get_lifecycle_manager() -> LifecycleManager:
    return _lifecycle_manager

def create_lifespan(app: FastAPI):
    """创建 FastAPI lifespan 上下文管理器."""
    
    @asynccontextmanager
    async def lifespan(app: FastAPI):
        # 启动
        await _lifecycle_manager.startup()
        yield
        # 关闭
        await _lifecycle_manager.shutdown()
    
    return lifespan

def setup_signal_handlers():
    """设置信号处理器."""
    loop = asyncio.get_event_loop()
    
    def signal_handler(sig, frame):
        logger.info(f"Received signal {sig}, initiating graceful shutdown...")
        asyncio.create_task(_lifecycle_manager.shutdown())
    
    for sig in (signal.SIGTERM, signal.SIGINT):
        try:
            loop.add_signal_handler(sig, signal_handler, sig, None)
        except NotImplementedError:
            # Windows 不支持
            pass
```

## 验收标准

| 探针类型 | 检查内容 | 启动时间目标 | 故障检测时间 |
|----------|----------|--------------|--------------|
| **Startup** | 关键依赖 (PG/Redis) 连通性 | <30s | 5s * 30 = 150s 后重启 |
| **Liveness** | 进程存活 | 即时 | 10s * 3 = 30s 后重启 |
| **Readiness** | 所有依赖 + 资源状态 | 启动后 | 5s * 3 = 15s 后摘除流量 |

| 指标 | 目标 | 实现 |
|------|------|------|
| 启动时间 | <30s | ✅ Startup Probe 并行检查 |
| 故障自愈 | <60s | ✅ Liveness 30s + Readiness 15s |
| 零停机部署 | 支持 | ✅ RollingUpdate + Readiness |
| 优雅关闭 | 30s 超时 | ✅ LifecycleManager |
| 依赖检查并行 | 支持 | ✅ asyncio.gather |
