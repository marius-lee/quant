# Phase 9.5 完成归档 — SLO/SLI 运维大盘

## 变更摘要
构建业务级 SLO/SLI 监控体系：可用性/延迟/数据新鲜度/同步延迟 SLO 定义、Burn Rate 多窗口告警、Grafana SLO 仪表盘、Error Budget 可视化。

---

## 核心概念

### SLI (Service Level Indicators) - 服务水平指标
| SLI 名称 | 定义 | PromQL 查询 |
|----------|------|-------------|
| **可用性** | 成功请求 / 总请求 | `sum(rate(http_requests_total{status=~"2.."}[5m])) / sum(rate(http_requests_total[5m]))` |
| **延迟 P99** | 99 分位请求延迟 | `histogram_quantile(0.99, rate(http_request_duration_seconds_bucket[5m]))` |
| **数据新鲜度** | 最新数据时间 - 当前时间 | `time() - max(data_last_update_timestamp)` |
| **同步延迟** | CDC 同步完成 - 数据产生时间 | `max(cdc_sync_lag_seconds)` |
| **因子计算及时率** | 截止时间前完成因子 / 总因子 | `factor_completed_before_deadline / factor_total` |

### SLO (Service Level Objectives) - 服务水平目标

| SLO | 目标 | 窗口 | Error Budget |
|-----|------|------|--------------|
| **API 可用性** | 99.9% | 30 天 | 43.2 分钟/月 |
| **API 延迟 P99** | < 500ms | 30 天 | - |
| **数据新鲜度** | < 1 小时 | 1 天 | - |
| **同步延迟** | < 5 分钟 | 1 天 | - |
| **因子计算及时率** | 99.5% | 1 天 | - |

### Error Budget Burn Rate - 燃尽率告警

| 严重级别 | 燃尽率 | 窗口 | 含义 |
|----------|--------|------|------|
| **Critical** | 14.4x | 1 小时 | 1 小时内耗尽 14.4 小时预算 → 立即响应 |
| **Warning** | 6x | 6 小时 | 6 小时内耗尽 6 小时预算 → 1 小时内响应 |
| **Info** | 1x | 24 小时 | 正常消耗速度 → 观察 |

### 2. SLO 监控器 (`quant/monitoring/slo_monitor.py`)

```python
# quant/monitoring/slo_monitor.py
"""SLO 监控器 - 计算 SLI、Error Budget、Burn Rate."""

from __future__ import annotations
import yaml
import asyncio
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from pathlib import Path
from typing import Dict, List, Optional, Any

from prometheus_client import Gauge, Histogram
from quant.utils.logger import get_logger
from quant.config.constants import _require_cfg

logger = get_logger("monitoring.slo_monitor")

class AlertSeverity(Enum):
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"

@dataclass
class SLOConfig:
    name: str
    display_name: str
    description: str
    sli: Dict[str, Any]
    target: float
    window: str
    alerting: Dict[str, Any]

@dataclass
class SLOStatus:
    name: str
    current_value: float
    target: float
    error_budget_remaining: float  # 0.0 - 1.0
    error_budget_consumed: float   # 0.0 - 1.0
    burn_rate: Dict[str, float]    # window -> burn_rate
    alert_status: AlertSeverity
    last_updated: datetime

class SLOMonitor:
    """SLO 监控器."""
    
    def __init__(self, config_path: str = "config/slo/slo-config.yaml"):
        self.config_path = Path(config_path)
        self.slos: Dict[str, SLOConfig] = {}
        self.status: Dict[str, SLOStatus] = {}
        self._prometheus_url = _require_cfg("prometheus.url", "http://prometheus:9090")
        self._load_config()
        
        # Prometheus 指标
        self.sli_gauge = Gauge('slo_sli_value', 'Current SLI value', ['slo_name'])
        self.error_budget_gauge = Gauge('slo_error_budget_remaining', 'Error budget remaining ratio', ['slo_name'])
        self.burn_rate_gauge = Gauge('slo_burn_rate', 'Error budget burn rate', ['slo_name', 'window'])
    
    def _load_config(self):
        with open(self.config_path) as f:
            data = yaml.safe_load(f)
        for slo_data in data.get('slos', []):
            slo = SLOConfig(**slo_data)
            self.slos[slo.name] = slo
    
    async def query_prometheus(self, query: str) -> float:
        """查询 Prometheus."""
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"{self._prometheus_url}/api/v1/query",
                params={"query": query}
            ) as resp:
                data = await resp.json()
                if data['status'] == 'success' and data['data']['result']:
                    return float(data['data']['result'][0]['value'][1])
                return 0.0
    
    async def evaluate_slo(self, slo: SLOConfig) -> SLOStatus:
        """评估单个 SLO."""
        sli = slo.sli
        
        if sli['type'] == 'ratio':
            good = await self.query_prometheus(sli['good_query'])
            total = await self.query_prometheus(sli['total_query'])
            current = good / total if total > 0 else 1.0
        elif sli['type'] == 'histogram':
            current = await self.query_prometheus(sli['query'])
        elif sli['type'] == 'gauge':
            current = await self.query_prometheus(sli['query'])
        else:
            current = 0.0
        
        # 计算 Error Budget
        if sli['type'] in ('ratio', 'histogram'):
            # ratio: 越高越好 / histogram: 越低越好
            if sli['type'] == 'ratio':
                error_budget_consumed = max(0, (slo.target - current) / (1 - slo.target)) if slo.target < 1 else 0
            else:
                error_budget_consumed = max(0, (current - slo.target) / slo.target) if slo.target > 0 else 0
        else:
            error_budget_consumed = max(0, (current - slo.target) / slo.target) if slo.target > 0 else 0
        
        error_budget_remaining = max(0, 1 - error_budget_consumed)
        
        # 计算 Burn Rate (简化版)
        burn_rate = {}
        for alert in slo.alerting.get('burn_rate', []):
            window = alert['window']
            # 实际应查询不同时间窗口的错误率
            # 这里简化：使用当前错误率 / 目标错误率
            target_error_rate = 1 - slo.target if sli['type'] == 'ratio' else slo.target
            current_error_rate = 1 - current if sli['type'] == 'ratio' else current
            burn_rate[window] = current_error_rate / target_error_rate if target_error_rate > 0 else 0
        
        # 确定告警状态
        alert_status = AlertSeverity.INFO
        for alert in slo.alerting.get('burn_rate', []):
            if burn_rate.get(alert['window'], 0) >= alert['threshold']:
                alert_status = AlertSeverity(alert['severity'])
                break
        for alert in slo.alerting.get('threshold', []):
            if current >= alert['value']:
                alert_status = AlertSeverity(alert['severity'])
                break
        
        return SLOStatus(
            name=slo.name,
            current_value=current,
            target=slo.target,
            error_budget_remaining=error_budget_remaining,
            error_budget_consumed=error_budget_consumed,
            burn_rate=burn_rate,
            alert_status=alert_status,
            last_updated=datetime.utcnow()
        )
    
    async def evaluate_all(self) -> Dict[str, SLOStatus]:
        """评估所有 SLO."""
        results = {}
        for name, slo in self.slos.items():
            try:
                status = await self.evaluate_slo(slo)
                results[name] = status
                
                # 更新 Prometheus 指标
                self.sli_gauge.labels(slo_name=name).set(status.current_value)
                self.error_budget_gauge.labels(slo_name=name).set(status.error_budget_remaining)
                for window, rate in status.burn_rate.items():
                    self.burn_rate_gauge.labels(slo_name=name, window=window).set(rate)
                    
            except Exception as e:
                logger.error(f"Failed to evaluate SLO {name}: {e}")
        self.status = results
        return results
    
    def get_status(self, name: str) -> Optional[SLOStatus]:
        return self.status.get(name)
    
    def get_all_status(self) -> Dict[str, SLOStatus]:
        return self.status

# 全局实例
_slo_monitor: Optional[SLOMonitor] = None

def get_slo_monitor() -> SLOMonitor:
    global _slo_monitor
    if _slo_monitor is None:
        _slo_monitor = SLOMonitor()
    return _slo_monitor
```

### 4. Alertmanager Burn Rate 告警规则 (`config/prometheus/rules/slo-alerts.yml`)

  - name: slo-alerts
    interval: 30s
    rules:
      # API 可用性 Burn Rate 告警
      - alert: SLOApiAvailabilityBurnRateCritical

          slo_burn_rate{slo_name="api_availability",window="1h"} > 14.4
        for: 2m
        labels:
          severity: critical
          slo: api_availability

          summary: "SLO: API 可用性 Error Budget 快速燃尽"
