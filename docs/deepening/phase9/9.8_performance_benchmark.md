# quant/benchmark/__init__.py
"""性能基准测试框架."""

from __future__ import annotations
import asyncio
import json
import statistics
import time
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from pathlib import Path
from typing import Any, Callable, Dict, List, Optional

from quant.utils.logger import get_logger

logger = get_logger("benchmark")

class BenchmarkType(Enum):
    THROUGHPUT = "throughput"      # 吞吐量测试
    LATENCY = "latency"            # 延迟测试
    STRESS = "stress"              # 压力测试
    SOAK = "soak"                  # 稳定性/长时间运行
    SPIKE = "spike"                # 突发流量测试

@dataclass
class BenchmarkConfig:
    name: str
    type: BenchmarkType
    duration_seconds: int = 60
    concurrency: int = 10
    ramp_up_seconds: int = 10
    target_qps: Optional[int] = None
    warmup_seconds: int = 30
    cooldown_seconds: int = 10

@dataclass
class BenchmarkResult:
    name: str
    config: BenchmarkConfig
    started_at: datetime
    completed_at: datetime
    total_requests: int = 0
    successful_requests: int = 0
    failed_requests: int = 0
    latencies_ms: List[float] = field(default_factory=list)
    errors: Dict[str, int] = field(default_factory=dict)
    throughput_qps: float = 0.0
    p50_ms: float = 0.0
    p95_ms: float = 0.0
    p99_ms: float = 0.0
    max_ms: float = 0.0
    error_rate: float = 0.0

    def calculate_percentiles(self):
        if not self.latencies_ms:
            return
        sorted_latencies = sorted(self.latencies_ms)
        n = len(sorted_latencies)
        self.p50_ms = sorted_latencies[int(n * 0.50)]
        self.p95_ms = sorted_latencies[int(n * 0.95)]
        self.p99_ms = sorted_latencies[int(n * 0.99)]
        self.max_ms = sorted_latencies[-1]
        self.throughput_qps = self.successful_requests / (self.completed_at - self.started_at).total_seconds()
        self.error_rate = self.failed_requests / max(self.total_requests, 1)

class BenchmarkRunner:
    """基准测试运行器."""
    
    def __init__(self, output_dir: Path = Path("benchmark_results")):
        self.output_dir = output_dir
        self.output_dir.mkdir(parents=True, exist_ok=True)
    
    async def run(
        self,
        config: BenchmarkConfig,
        request_fn: Callable[[], Any],
        setup_fn: Optional[Callable] = None,
        teardown_fn: Optional[Callable] = None,
    ) -> BenchmarkResult:
        """运行基准测试."""
        result = BenchmarkResult(
            name=config.name,
            config=config,
            started_at=datetime.utcnow(),
            completed_at=datetime.utcnow(),  # 稍后更新
        )
        
        logger.info(f"Starting benchmark: {config.name} ({config.type.value})")
        
        # Setup
        if setup_fn:
            await setup_fn()
        
        # Warmup
        if config.warmup_seconds > 0:
            logger.info(f"Warming up for {config.warmup_seconds}s...")
            await self._run_phase(config, request_fn, config.warmup_seconds, result, is_warmup=True)
        
        # Main test
        logger.info(f"Running benchmark for {config.duration_seconds}s...")
        await self._run_phase(config, request_fn, config.duration_seconds, result)
        
        # Cooldown
        if config.cooldown_seconds > 0:
            logger.info(f"Cooling down for {config.cooldown_seconds}s...")
            await asyncio.sleep(config.cooldown_seconds)
        
        # Teardown
        if teardown_fn:
            await teardown_fn()
        
        result.completed_at = datetime.utcnow()
        result.calculate_percentiles()
        
        # 保存结果
        self._save_result(result)
        
        logger.info(f"Benchmark {config.name} completed: {result.throughput_qps:.2f} QPS, P99={result.p99_ms:.1f}ms")
        return result
    
    async def _run_phase(
        self,
        config: BenchmarkConfig,
        request_fn: Callable,
        duration: int,
        result: BenchmarkResult,
        is_warmup: bool = False
    ):
        """运行单个阶段."""
        semaphore = asyncio.Semaphore(config.concurrency)
        start_time = time.time()
        end_time = start_time + duration
        
        async def worker():
            while time.time() < end_time:
                async with semaphore:
                    req_start = time.perf_counter()
                    try:
                        await request_fn()
                        latency = (time.perf_counter() - req_start) * 1000
                        if not is_warmup:
                            result.total_requests += 1
                            result.successful_requests += 1
                            result.latencies_ms.append(latency)
                    except Exception as e:
                        if not is_warmup:
                            result.total_requests += 1
                            result.failed_requests += 1
                            error_type = type(e).__name__
                            result.errors[error_type] = result.errors.get(error_type, 0) + 1
                    # 控制 QPS
                    if config.target_qps:
                        await asyncio.sleep(1.0 / config.target_qps)
        
        # 启动 worker
        workers = [asyncio.create_task(worker()) for _ in range(config.concurrency)]
        await asyncio.gather(*workers)
    
    def _save_result(self, result: BenchmarkResult):
        """保存结果到文件."""
        output_file = self.output_dir / f"{result.name}_{result.started_at.strftime('%Y%m%d_%H%M%S')}.json"
        with open(output_file, 'w') as f:
            json.dump({
                "name": result.name,
                "config": {
                    "type": result.config.type.value,
                    "duration_seconds": result.config.duration_seconds,
                    "concurrency": result.config.concurrency,
                    "target_qps": result.config.target_qps,
                },
                "started_at": result.started_at.isoformat(),
                "completed_at": result.completed_at.isoformat(),
                "total_requests": result.total_requests,
                "successful_requests": result.successful_requests,
                "failed_requests": result.failed_requests,
                "throughput_qps": result.throughput_qps,
                "p50_ms": result.p50_ms,
                "p95_ms": result.p95_ms,
                "p99_ms": result.p99_ms,
                "max_ms": result.max_ms,
                "error_rate": result.error_rate,
                "errors": result.errors,
            }, f, indent=2)
        logger.info(f"Result saved to {output_file}")

# 便捷函数
async def run_throughput_benchmark(
    name: str,
    request_fn: Callable,
    duration: int = 60,
    concurrency: int = 10,
    target_qps: Optional[int] = None,
) -> BenchmarkResult:
    runner = BenchmarkRunner()
    config = BenchmarkConfig(
        name=name,
        type=BenchmarkType.THROUGHPUT,
        duration_seconds=duration,
        concurrency=concurrency,
        target_qps=target_qps,
    )
    return await runner.run(config, request_fn)

async def run_latency_benchmark(
    name: str,
    request_fn: Callable,
    duration: int = 60,
    concurrency: int = 1,
) -> BenchmarkResult:
    runner = BenchmarkRunner()
    config = BenchmarkConfig(
        name=name,
        type=BenchmarkType.LATENCY,
        duration_seconds=duration,
        concurrency=concurrency,
    )
    return await runner.run(config, request_fn)

async def run_stress_benchmark(
    name: str,
    request_fn: Callable,
    duration: int = 300,
    max_concurrency: int = 100,
    step: int = 10,
    step_duration: int = 60,
) -> List[BenchmarkResult]:
    """阶梯式压力测试."""
    results = []
    for concurrency in range(step, max_concurrency + 1, step):
        config = BenchmarkConfig(
            name=f"{name}_c{concurrency}",
            type=BenchmarkType.STRESS,
            duration_seconds=step_duration,
            concurrency=concurrency,
        )
        runner = BenchmarkRunner()
        result = await runner.run(config, request_fn)
        results.append(result)
        
        # 如果错误率过高，提前停止
        if result.error_rate > 0.05:
            logger.warning(f"Error rate {result.error_rate:.2%} > 5%, stopping stress test")
            break
    
    return results

# quant/benchmark/capacity.py
"""容量规划模型."""

from __future__ import annotations
from dataclasses import dataclass
from typing import Dict, List

@dataclass
class ResourceRequirement:
    cpu_cores: float
    memory_gb: float
    storage_gb: float
    network_mbps: float

@dataclass
class CapacityPlan:
    scenario: str
    current_qps: float
    target_qps: float
    current_resources: ResourceRequirement
    required_resources: ResourceRequirement
    scaling_factor: float
    recommended_instances: int

def calculate_capacity(
    benchmark_results: Dict,
    target_load: Dict[str, float],
    current_instances: Dict[str, int],
    instance_spec: Dict[str, Dict],  # {component: {cpu, memory}}
) -> List[CapacityPlan]:
    """根据基准测试结果计算容量需求."""
    plans = []
    
    for scenario, result in benchmark_results.items():
        current_qps = result.get("throughput_qps") or result.get("throughput_rows_per_sec") or 0
        target_qps = target_load.get(scenario, current_qps * 2)
        
        if current_qps == 0:
            continue
        
        scaling_factor = target_qps / current_qps
        
        # 获取当前组件资源
        component = scenario.split("_")[0]  # data, factor, risk, order
        spec = instance_spec.get(component, {"cpu": 4, "memory": 8})
        
        current_cpu = spec["cpu"] * current_instances.get(component, 1)
        current_mem = spec["memory"] * current_instances.get(component, 1)
        
        required_cpu = current_cpu * scaling_factor
        required_mem = current_mem * scaling_factor
        
        # 计算需要的实例数 (向上取整，预留 20% 余量)
        required_instances = max(1, int((required_cpu / spec["cpu"]) * 1.2))
        
        plans.append(CapacityPlan(
            scenario=scenario,
            current_qps=current_qps,
            target_qps=target_qps,
            current_resources=ResourceRequirement(
                cpu_cores=current_cpu,
                memory_gb=current_mem,
                storage_gb=100,
                network_mbps=1000,
            ),
            required_resources=ResourceRequirement(
                cpu_cores=required_cpu,
                memory_gb=required_mem,
                storage_gb=200,
                network_mbps=2000,
            ),
            scaling_factor=scaling_factor,
            recommended_instances=required_instances,
        ))
    
    return plans

# 示例用法
if __name__ == "__main__":
    # 模拟基准测试结果
    benchmark_results = {
        "data_ingestion": {"throughput_rows_per_sec": 500000},
        "factor_computation": {"throughput_qps": 100},
        "realtime_risk": {"p99_latency_ms": 5},
        "order_execution": {"throughput_qps": 2000},
    }
    
    target_load = {
        "data_ingestion": 2000000,  # 200万行/秒
        "factor_computation": 500,   # 500 因子/秒
        "realtime_risk": 1,          # 延迟测试
        "order_execution": 10000,    # 1万单/秒
    }
    
    current_instances = {
        "data": 3,
        "factor": 6,
        "risk": 2,
        "order": 4,
    }
    
    instance_spec = {
        "data": {"cpu": 8, "memory": 16},
        "factor": {"cpu": 16, "memory": 32},
        "risk": {"cpu": 8, "memory": 16},
        "order": {"cpu": 8, "memory": 16},
    }
    
    plans = calculate_capacity(benchmark_results, target_load, current_instances, instance_spec)
    
    for plan in plans:
        print(f"\n{plan.scenario}:")
        print(f"  Current: {plan.current_qps:.0f} QPS → Target: {plan.target_qps:.0f} QPS")
        print(f"  Scaling factor: {plan.scaling_factor:.2f}x")
        print(f"  Recommended instances: {plan.recommended_instances}")
        print(f"  CPU: {plan.current_resources.cpu_cores:.0f} → {plan.required_resources.cpu_cores:.0f} cores")
        print(f"  Memory: {plan.current_resources.memory_gb:.0f} → {plan.required_resources.memory_gb:.0f} GB")

### 5. 性能回归对比脚本

```python
# scripts/compare_benchmarks.py
"""性能回归对比脚本."""

import argparse
import json
import sys
from typing import Dict, Any

def load_results(filepath: str) -> Dict[str, Any]:
    with open(filepath) as f:
        return json.load(f)

def compare_scenario(name: str, current: Dict, baseline: Dict, threshold: float) -> Dict:
    """对比单个场景."""
    metrics = ["throughput_qps", "throughput_rows_per_sec", "p99_latency_ms", "p50_latency_ms"]
    regressions = []
    
    for metric in metrics:
        if metric not in current or metric not in baseline:
            continue
        
        curr_val = current[metric]
        base_val = baseline[metric]
        
        if base_val == 0:
            continue
        
        # 延迟指标：越低越好
        if "latency" in metric:
            change_pct = (curr_val - base_val) / base_val
            if change_pct > threshold:
                regressions.append(f"{metric}: {base_val:.1f} → {curr_val:.1f} ({change_pct:.1%} 较慢)")
        # 吞吐指标：越高越好
        else:
            change_pct = (base_val - curr_val) / base_val
            if change_pct > threshold:
                regressions.append(f"{metric}: {base_val:.1f} → {curr_val:.1f} ({change_pct:.1%} 下降)")
    
    return {
        "scenario": name,
        "regressions": regressions,
        "passed": len(regressions) == 0,
    }

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--current", required=True)
    parser.add_argument("--baseline", required=True)
    parser.add_argument("--threshold", type=float, default=0.10)
    args = parser.parse_args()
    
    current = load_results(args.current)
    baseline = load_results(args.baseline)
    
    all_passed = True
    for scenario in current:
        if scenario not in baseline:
            print(f"⚠️  {scenario}: No baseline data")
            continue
        
        result = compare_scenario(scenario, current[scenario], baseline[scenario], args.threshold)
        
        if result["passed"]:
            print(f"✅ {scenario}: No regression")
        else:
            print(f"❌ {scenario}: REGRESSION DETECTED")
            for reg in result["regressions"]:
                print(f"   - {reg}")
            all_passed = False
    
    if not all_passed:
        print("\n🚨 Performance regression detected!")
        sys.exit(1)
    else:
        print("\n✅ All benchmarks passed regression check")
        sys.exit(0)

if __name__ == "__main__":
    main()
```
