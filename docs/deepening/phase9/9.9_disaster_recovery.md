建立完整的灾难恢复体系：ETCD/PostgreSQL/DuckDB/MinIO 备份恢复、RPO<5min/RTO<30min、自动化恢复脚本、混沌工程验证。

## 核心实现

### 1. 备份策略矩阵

| 数据存储 | 备份方式 | 频率 | 保留 | RPO | RTO | 验证频率 |
|----------|----------|------|------|-----|-----|----------|
| **PostgreSQL** | pg_basebackup + WAL-G | 持续 (WAL) + 每日全量 | 30 天 | < 1 min | < 15 min | 每周 |
| **Redis** | RDB + AOF | 每 1 小时 | 7 天 | < 1 min | < 5 min | 每周 |
| **MinIO** | mc mirror (跨区域) | 每 6 小时 | 90 天 | < 5 min | < 10 min | 每月 |
| **DuckDB** | 文件级复制 | 每日增量 + 周全量 | 30 天 | < 5 min | < 10 min | 每周 |
| **ETCD** | etcdctl snapshot | 每 30 分钟 | 7 天 | < 30 sec | < 5 min | 每周 |
| **Kubernetes** | Velero (资源+PV) | 每 4 小时 | 30 天 | < 4 小时 | < 30 min | 每月 |

# quant/scripts/backup.py
"""统一备份编排脚本."""

from __future__ import annotations
import asyncio
import gzip
import os
import shutil
import subprocess
import time
from datetime import datetime, timedelta
from pathlib import Path
from typing import Dict, List, Optional

from quant.utils.logger import get_logger

logger = get_logger("backup")

class BackupOrchestrator:
    """备份编排器."""
    
    def __init__(self, backup_root: str = "/backups"):
        self.backup_root = Path(backup_root)
        self.backup_root.mkdir(parents=True, exist_ok=True)
        self.results: Dict[str, Dict] = {}
    
    async def run_all_backups(self) -> Dict:
        """执行所有备份任务."""
        tasks = [
            self.backup_postgresql(),
            self.backup_redis(),
            self.backup_minio(),
            self.backup_duckdb(),
            self.backup_etcd(),
            self.backup_kubernetes(),
        ]
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        for i, result in enumerate(results):
            task_name = ["postgresql", "redis", "minio", "duckdb", "etcd", "kubernetes"][i]
            if isinstance(result, Exception):
                self.results[task_name] = {"success": False, "error": str(result)}
                logger.error(f"Backup {task_name} failed: {result}")
            else:
                self.results[task_name] = result
        
        return self.results
    
    # ═══════════════════════════════════════════════════════════
    # PostgreSQL: pg_basebackup + WAL-G
    # ═══════════════════════════════════════════════════════════
    async def backup_postgresql(self) -> Dict:
        """PostgreSQL 备份: 基础备份 + WAL 归档."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        backup_dir = self.backup_root / "postgresql" / timestamp
        backup_dir.mkdir(parents=True, exist_ok=True)
        
        # 1. 基础备份
        cmd = [
            "pg_basebackup",
            "-h", os.getenv("POSTGRES_HOST", "localhost"),
            "-p", os.getenv("POSTGRES_PORT", "5432"),
            "-U", os.getenv("POSTGRES_USER", "quant"),
            "-D", str(backup_dir),
            "-Ft",  # tar 格式
            "-z",   # 压缩
            "-P",   # 进度
            "-X", "stream",  # 包含 WAL
        ]
        
        env = os.environ.copy()
        env["PGPASSWORD"] = os.getenv("POSTGRES_PASSWORD", "")
        
        start = time.time()
        proc = await asyncio.create_subprocess_exec(
            *cmd,
            env=env,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        stdout, stderr = await proc.communicate()
        duration = time.time() - start
        
        if proc.returncode != 0:
            raise RuntimeError(f"pg_basebackup failed: {stderr.decode()}")
        
        # 2. 压缩打包
        tar_file = backup_dir.with_suffix(".tar.gz")
        shutil.make_archive(str(backup_dir), 'gztar', backup_dir)
        shutil.rmtree(backup_dir)
        
        # 3. 清理旧备份 (保留 30 天)
        await self._cleanup_old(self.backup_root / "postgresql", days=30)
        
        size = tar_file.stat().st_size
        return {
            "success": True,
            "file": str(tar_file),
            "size_bytes": size,
            "duration_seconds": duration,
        }
    
    # ═══════════════════════════════════════════════════════════
    # Redis: RDB + AOF
    # ═══════════════════════════════════════════════════════════
    async def backup_redis(self) -> Dict:
        """Redis 备份: BGSAVE + 复制 RDB/AOF."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        backup_dir = self.backup_root / "redis" / timestamp
        backup_dir.mkdir(parents=True, exist_ok=True)
        
        # 触发 BGSAVE
        import redis
        client = redis.from_url(os.getenv("REDIS_URL", "redis://localhost:6379"))
        client.bgsave()
        
        # 等待完成
        while True:
            info = client.info("persistence")
            if info.get("rdb_bgsave_in_progress") == 0:
                break
            await asyncio.sleep(1)
        
        # 复制 RDB 文件
        rdb_path = Path("/data/dump.rdb")  # Redis 数据目录
        if rdb_path.exists():
            dest = backup_dir / "dump.rdb"
            shutil.copy2(rdb_path, dest)
        
        # 复制 AOF 文件
        aof_path = Path("/data/appendonly.aof")
        if aof_path.exists():
            dest = backup_dir / "appendonly.aof"
            shutil.copy2(aof_path, dest)
        
        # 压缩
        tar_file = backup_dir.with_suffix(".tar.gz")
        shutil.make_archive(str(backup_dir), 'gztar', backup_dir)
        shutil.rmtree(backup_dir)
        
        await self._cleanup_old(self.backup_root / "redis", days=7)
        
        size = tar_file.stat().st_size
        return {"success": True, "file": str(tar_file), "size_bytes": size}
    
    # ═══════════════════════════════════════════════════════════
    # MinIO: mc mirror 跨区域复制
    # ═══════════════════════════════════════════════════════════
    async def backup_minio(self) -> Dict:
        """MinIO 备份: mc mirror 到远程存储."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        
        # 假设配置了远程 MinIO 别名 "backup"
        buckets = ["quant-data", "quant-models", "quant-reports"]
        
        for bucket in buckets:
            cmd = [
                "mc", "mirror",
                "--overwrite",
                "--remove",
                f"local/{bucket}",
                f"backup/{bucket}",
            ]
            
            proc = await asyncio.create_subprocess_exec(
                *cmd,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            await proc.communicate()
            
            if proc.returncode != 0:
                logger.warning(f"MinIO mirror {bucket} completed with warnings")
        
        return {"success": True, "buckets": buckets}
    
    # ═══════════════════════════════════════════════════════════
    # DuckDB: 文件级增量备份
    # ═══════════════════════════════════════════════════════════
    async def backup_duckdb(self) -> Dict:
        """DuckDB 备份: 文件复制 + 压缩."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        backup_dir = self.backup_root / "duckdb" / timestamp
        backup_dir.mkdir(parents=True, exist_ok=True)
        
        source = Path("/data/market.duckdb")
        if source.exists():
            dest = backup_dir / "market.duckdb"
            shutil.copy2(source, dest)
            
            # 压缩
            tar_file = backup_dir.with_suffix(".tar.gz")
            shutil.make_archive(str(backup_dir), 'gztar', backup_dir)
            shutil.rmtree(backup_dir)
            
            await self._cleanup_old(self.backup_root / "duckdb", days=30)
            
            size = tar_file.stat().st_size
            return {"success": True, "file": str(tar_file), "size_bytes": size}
        else:
            return {"success": False, "error": "DuckDB file not found"}
    
    # ═══════════════════════════════════════════════════════════
    # ETCD: etcdctl snapshot
    # ═══════════════════════════════════════════════════════════
    async def backup_etcd(self) -> Dict:
        """ETCD 备份: etcdctl snapshot save."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        backup_dir = self.backup_root / "etcd" / timestamp
        backup_dir.mkdir(parents=True, exist_ok=True)
        
        snapshot_file = backup_dir / "etcd-snapshot.db"
        
        cmd = [
            "etcdctl", "snapshot", "save", str(snapshot_file),
            "--endpoints", os.getenv("ETCD_ENDPOINTS", "https://etcd:2379"),
            "--cacert", "/etc/etcd/ca.crt",
            "--cert", "/etc/etcd/client.crt",
            "--key", "/etc/etcd/client.key",
        ]
        
        proc = await asyncio.create_subprocess_exec(
            *cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        await proc.communicate()
        
        if proc.returncode != 0:
            raise RuntimeError("etcdctl snapshot failed")
        
        await self._cleanup_old(self.backup_root / "etcd", days=7)
        
        size = snapshot_file.stat().st_size
        return {"success": True, "file": str(snapshot_file), "size_bytes": size}
    
    # ═══════════════════════════════════════════════════════════
    # Kubernetes: Velero
    # ═══════════════════════════════════════════════════════════
    async def backup_kubernetes(self) -> Dict:
        """Kubernetes 备份: Velero backup."""
        timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
        backup_name = f"quant-full-{timestamp}"
        
        cmd = [
            "velero", "backup", "create", backup_name,
            "--include-namespaces", "quant-prod,quant-staging,monitoring",
            "--wait",
        ]
        
        proc = await asyncio.create_subprocess_exec(
            *cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        stdout, stderr = await proc.communicate()
        
        if proc.returncode != 0:
            logger.warning(f"Velero backup completed with warnings: {stderr.decode()}")
        
        return {"success": True, "backup_name": backup_name}
    
    async def _cleanup_old(self, backup_dir: Path, days: int):
        """清理旧备份."""
        cutoff = time.time() - (days * 86400)
        for item in backup_dir.iterdir():
            if item.is_file() and item.stat().st_mtime < cutoff:
                item.unlink()
                logger.info(f"Deleted old backup: {item}")

# 定时任务入口
async def main():
    orchestrator = BackupOrchestrator()
    results = await orchestrator.run_all_backups()
    
    # 汇总报告
    print("\n" + "="*60)
    print("BACKUP SUMMARY")
    print("="*60)
    for name, result in results.items():
        status = "✅" if result.get("success") else "❌"
        print(f"{status} {name}: {result}")
    
    # 发送通知
    failed = [k for k, v in results.items() if not v.get("success")]
    if failed:
        # 发送告警
        pass

if __name__ == "__main__":
    asyncio.run(main())

from __future__ import annotations
import asyncio
import random
from datetime import datetime
from enum import Enum
from typing import Dict, List, Optional

from quant.utils.logger import get_logger

logger = get_logger("chaos")

class ChaosExperiment(Enum):
    POD_KILL = "pod_kill"           # 随机杀 Pod
    NETWORK_PARTITION = "network_partition"  # 网络分区
    CPU_STRESS = "cpu_stress"       # CPU 压力
    MEMORY_STRESS = "memory_stress" # 内存压力
    DISK_FILL = "disk_fill"         # 磁盘填满
    DNS_FAILURE = "dns_failure"     # DNS 解析失败
    LATENCY_INJECTION = "latency_injection"  # 延迟注入

class ChaosRunner:
    """混沌实验运行器."""
    
    def __init__(self, namespace: str = "quant-prod"):
        self.namespace = namespace
        self.active_experiments: Dict[str, asyncio.Task] = {}
    
    async def run_experiment(
        self,
        experiment: ChaosExperiment,
        target: str,  # deployment name or label selector
        duration_seconds: int = 60,
        intensity: float = 0.5,  # 0.0 - 1.0
    ) -> Dict:
        """运行混沌实验."""
        logger.warning(f"Starting chaos experiment: {experiment.value} on {target}")
        
        experiment_id = f"{experiment.value}_{target}_{datetime.utcnow().strftime('%Y%m%d_%H%M%S')}"
        
        if experiment == ChaosExperiment.POD_KILL:
            task = asyncio.create_task(self._pod_kill(target, duration_seconds, intensity))
        elif experiment == ChaosExperiment.CPU_STRESS:
            task = asyncio.create_task(self._cpu_stress(target, duration_seconds, intensity))
        elif experiment == ChaosExperiment.MEMORY_STRESS:
            task = asyncio.create_task(self._memory_stress(target, duration_seconds, intensity))
        elif experiment == ChaosExperiment.NETWORK_PARTITION:
            task = asyncio.create_task(self._network_partition(target, duration_seconds))
        elif experiment == ChaosExperiment.DISK_FILL:
            task = asyncio.create_task(self._disk_fill(target, duration_seconds, intensity))
        elif experiment == ChaosExperiment.LATENCY_INJECTION:
            task = asyncio.create_task(self._latency_injection(target, duration_seconds, intensity))
        else:
            raise ValueError(f"Unknown experiment: {experiment}")
        
        self.active_experiments[experiment_id] = task
        
        try:
            result = await asyncio.wait_for(task, timeout=duration_seconds + 30)
            return {"experiment_id": experiment_id, "success": True, "result": result}
        except asyncio.TimeoutError:
            task.cancel()
            return {"experiment_id": experiment_id, "success": False, "error": "Timeout"}
        except Exception as e:
            return {"experiment_id": experiment_id, "success": False, "error": str(e)}
        finally:
            self.active_experiments.pop(experiment_id, None)
    
    async def _pod_kill(self, target: str, duration: int, intensity: float) -> Dict:
        """随机杀 Pod."""
        import kubernetes.client as k8s_client
        from kubernetes import config
        
        config.load_incluster_config()
        v1 = k8s_client.CoreV1Api()
        
        # 获取目标 Pod
        pods = v1.list_namespaced_pod(self.namespace, label_selector=target)
        target_pods = [p for p in pods.items if p.status.phase == "Running"]
        
        if not target_pods:
            return {"error": "No running pods found"}
        
        # 计算杀掉数量
        kill_count = max(1, int(len(target_pods) * intensity))
        victims = random.sample(target_pods, kill_count)
        
        killed = []
        for pod in victims:
            v1.delete_namespaced_pod(pod.metadata.name, self.namespace)
            killed.append(pod.metadata.name)
            logger.warning(f"Killed pod: {pod.metadata.name}")
        
        # 等待恢复
        await asyncio.sleep(duration)
        
        # 验证恢复
        new_pods = v1.list_namespaced_pod(self.namespace, label_selector=target)
        recovered = all(p.status.phase == "Running" for p in new_pods.items)
        
        return {
            "killed": killed,
            "recovered": recovered,
            "duration_seconds": duration,
        }

# SOP: 灾难恢复演练

## 演练类型与频率

| 类型 | 范围 | 频率 | 参与者 | 持续时间 |
|------|------|------|--------|----------|
| **单组件恢复** | 单个数据存储 | 每月 | On-call + DBA | 1 小时 |
| **全链路恢复** | 全栈 (K8s + 数据) | 每季度 | 全团队 | 4 小时 |
| **跨区域故障转移** | 主备集群切换 | 半年 | 全团队 + 高管 | 8 小时 |

## 季度全链路演练流程

### 准备阶段 (T-1 周)
- [ ] 确认演练日期、时间、参与人员
- [ ] 准备演练环境 (独立命名空间/集群)
- [ ] 生成最新备份 (全组件)
- [ ] 通知相关方 (避免误判为真实事故)
- [ ] 准备观测大盘 (Grafana/Alertmanager 静音)

### 执行阶段 (演练日)

#### 场景 1: PostgreSQL 主库故障 (RTO 目标 < 15min)
```
T+0min   模拟: 删除 PostgreSQL 主 Pod + PVC
T+1min   告警触发: PostgreSQLDown
T+2min   On-call 响应，确认故障
T+5min   执行恢复: 从最新基础备份 + WAL 恢复
T+12min  PostgreSQL 就绪，只读验证
T+15min  服务恢复，流量切回
```
**验收**: 数据零丢失 (RPO=0), 服务恢复 < 15min

#### 场景 2: Kubernetes 控制平面故障 (RTO 目标 < 30min)
```
T+0min   模拟: 停止 API Server (或删除控制平面节点)
T+2min   集群不可管理，但工作负载继续运行
T+5min   启动备用控制平面 (或等待托管集群自愈)
T+15min  控制平面恢复
T+20min  验证所有 Controller 正常
T+30min  全栈验证通过
```
**验收**: 工作负载零影响, 管理面恢复 < 30min

#### 场景 3: 区域级故障 (RTO 目标 < 2小时)
```
T+0min   模拟: 整个可用区断电/网络分区
T+5min   确认主区域不可用
T+10min  启动 DR 区域: 修改 DNS/全局 LB
T+30min  DR 区域数据库提升为主
T+60min  应用部署到 DR 区域 (Velero restore)
T+90min  流量切换到 DR 区域
T+120min 全业务验证通过
```
**验收**: RPO < 5min (异步复制延迟), RTO < 2小时

# quant/scripts/dr_verify.py
"""灾难恢复自动化验证."""

from __future__ import annotations
import asyncio
import httpx
from dataclasses import dataclass
from typing import Dict, List

@dataclass
class VerificationResult:
    check_name: str
    passed: bool
    message: str
    latency_ms: float = 0

class DRVerifier:
    """灾难恢复验证器."""
    
    def __init__(self, base_url: str = "https://quant.example.com"):
        self.base_url = base_url
        self.client = httpx.AsyncClient(timeout=30.0)
    
    async def verify_all(self) -> Dict:
        """执行所有验证."""
        checks = [
            self.verify_api_health(),
            self.verify_data_ingestion(),
            self.verify_factor_computation(),
            self.verify_risk_engine(),
            self.verify_order_execution(),
            self.verify_monitoring(),
        ]
        
        results = await asyncio.gather(*checks, return_exceptions=True)
        
        passed = 0
        failed = 0
        details = []
        
        for i, result in enumerate(results):
            if isinstance(result, VerificationResult):
                if result.passed:
                    passed += 1
                else:
                    failed += 1
                details.append(result)
            else:
                failed += 1
                details.append(VerificationResult(
                    check_name=f"check_{i}",
                    passed=False,
                    message=str(result)
                ))
        
        return {
            "overall_passed": failed == 0,
            "passed": passed,
            "failed": failed,
            "details": [d.__dict__ for d in details],
        }

                return VerificationResult("api_health", True, "OK", latency)
            return VerificationResult("api_health", False, f"Status: {resp.status_code}", latency)
        except Exception as e:
            return VerificationResult("api_health", False, str(e), (time.perf_counter() - start) * 1000)
